                              1. civavaṇakkam

       āciriyaviruttam

   aṅkiṅkeṉātapaṭi yeṅkumprakācamā
             yāṉantapūrttiyāki
       yaruḷoṭuniṟaintatetu taṉṉaruḷveḷikkuḷē
             yakilāṇṭakōṭiyellān
   taṅkumpaṭikkiccai vaittuyirkkuyirāyt
             taḻaittatetumaṉavākkiṉiṟ
      ṟaṭṭāmaṉiṉṟatetu camayakōṭikaḷelān
             tanteyvamenteyvameṉ
   ṟeṅkuntoṭarntetir vaḻakkiṭavuniṉṟatetu
             veṅkaṇumperuvaḻkkā
      yātiṉumvallavoru cittākiyiṉpamā
             yeṉṟaikkumuḷḷatetumēṟa
   kaṅkulpakalaṟaniṉṟa vellaiyuḷtetuvatu
             karuttiṟkicaintatuvē
      kaṇṭaṉavelāmōṉa vuruveḷiyatākavuṅ
             karutiyañcali ceykuvām.      (1)
   ūraṉantampeṟṟa pēraṉantañcuṟṟu
             muṟavaṉantamviṉaiyiṉā
       luṭalaṉantañceyum viṉaiyaṉantaṅkarut
             tōvaṉantampeṟṟapēr
   cīraṉantañcorkka narakamumaṉantāṟ
             ṟeyvamumaṉantapētan
       tikaḻkiṉṟacamayamu maṉantamataṉāṉñāṉa
             ciṟcattiyāluṇarntu
   kāraṉantaṅkōṭi varuṣittateṉavaṉpar
             kaṇṇummiṇṇuntēkkavē
       karutariyavāṉanta maḻaipoḻiyumukilainaṅ
             kaṭavuḷaitturiyavaṭivaip
   pēraṉantampēci maṟaiyaṉantañcolum
             periyamavuṉattiṉvaippaip
       pēcarumaṉantapata ñāṉavāṉantamām
                 periyaporuḷaippaṇikuvām. (2)
   attuvitavattuvaic coṟprakācattaṉiyai
             yarumaṟaikaṇmuracaṟaiyavē
       yaṟiviṉukkaṟivāki yāṉantamayamāṉa
             vātiyaiyanātiyēka
   tattuvacorūpattai matacammatampeṟāc
             cālamparakitamāṉa
      cācuvatpuṭkala nirālampavālampa
             cāntapatavyōmanilaiyai
   nittanirmalacakita niṣprapañcapporuḷai
             nirviṣayacuttamāṉa
       nirvikārattait taṭattamāyniṉṟoḷir
             nirañcaṉanirāmayattaic
   cittamaṟiyātapaṭi cittattiṉiṉṟilaku
             tivyatēcōmayattaic
      ciṟparaveḷikkuḷvaḷar taṟparamatāṉapara
                 tēvataiyaiyañcaliceyvām  (3)
       ------------

                          2. paripūraṇāṉantam

   vācākayiṅkariya maṉṟiyorucātaṉa
             maṉōvāyuniṟkumvaṇṇam
   vālāyamākavum paḻakiyaṟiyēṉṟuṟavu
             mārkkattiṉiccaipōla
   nēcāṉucāriyāy vivakarippēṉanta
             niṉaivaiyumaṟantapōtu
   nittiraikoḷvēṉṟēka nīṅkumeṉaveṇṇilō
             neñcantuṭittayaruvēṉ
   pēcātavāṉanta niṭṭaikkkumaṟivilāp
             pētaikkumvekutūramē
   pēykkuṇamaṟintinta nāykkumoruvaḻiperiya
             pēriṉpaniṭṭaiyaruḷvāy
   pācāṭavikkuḷḷē cellātavarkkaruḷ
             paḻuttoḻukutēvataruvē
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (1)
   terivākavūrvaṉa naṭappaṉapaṟappaṉa
             ceyaṟkoṇṭiruppaṉa mutaṟ
   ṟēkaṅkaḷattaṉaiyu mōkaṅkoḷpautikañ
             ceṉmittavāṅkiṟakkum
   virivāyapūtaṅka ḷoṉṟoṭoṉṟāyaḻiyu
             mēṟkoṇṭacēṭamatuvē
   veṟuveḷinirālampa niṟaicūṉyamupacānta
             vētavētāntañāṉam
   piriyātapēroḷi piṟakkiṉṟavaruḷaruṭ
             peṟṟōrkaḷpeṟṟaperumai
   piṟavāmaiyeṉṟaikku miṟavāmaiyāyvantu
             pēcāmaiyākumeṉavē
   parivāyeṉakkunī yaṟivikkavantatē
             paripākakālamalavō
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (2)
   ārāyumvēḷaiyiṟ piramātiyāṉālu
             maiyavoruceyalumillai.
   yamaitiyoṭupēcāta perumaipeṟukuṇacantra
             rāmeṉaviruntapēru
   nērākavorukōpa moruvēḷaivaravanta
             niṟaivoṉṟumillāmalē
   neṭṭuyirttuttaṭ ṭaḻintuḷaṟuvārvacaṉa
             nirvākareṉṟapērum
   pūrāyamāyoṉṟu pēcumiṭamoṉṟaip
             pulampuvārcivarāttirip
   pōtutuyilōmeṉṟa viratiyarumaṟituyiṟ
             pōlēyiruntutuyilvār
   pārātitaṉiluḷḷa ceyalelāmuṭivilē
             pārkkiṉiṉceyalallavō
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (3)
   aṇṭapakiraṇṭamum māyāvikāramē
             ymmāyaiyillāmaiyē
   yāmeṉavumaṟivumuṇ ṭappālumaṟikiṉṟa
             vaṟiviṉaiyaṟintupārkki
   ṉeṇṭicaiviḷakkumoru teyvavaruḷallām
             lillaiyeṉuniṉaivumuṇṭiṅ
   kiyāṉeṉataṟatturiya niṟaivākiniṟpatē
             yiṉpameṉumaṉpumuṇṭu
   kaṇṭaṉavelāmalla veṉṟukaṇṭaṉaiceytu
             karulikaraṇaṅkaḷōyak
   kaṇmūṭiyorukaṇa mirukkaveṉṟāṟpāḻtta
             karmaṅkaḷpōrāṭutē
   paṇṭaiyuḷakarmamē karttāveṉumpeyarp
             pakṣanāṉiccippaṉō
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             pripūraṇāṉantamē. (4)
   cantatamumeṉatuceya ṉiṉatuceyaliyāṉeṉun
             taṉmainiṉaiyaṉṟiyillāt
   taṉmaiyālvēṟalēṉ vētāntacittānta
             camaracacupāvamituvē
   yintanilaiteḷiyanā ṉekkurukivāṭiya
             viyaṟkaitiruvuḷamaṟiyumē
   yinnilaiyilēcaṟ ṟirukkaveṉṟāṉmaṭamai
             yitacatruvākavantu
   cintaikuṭikoḷḷutē malamāyaikaṉman
             tirumpumōtoṭuvaḻakkāyc
   ceṉmamvarumōveṉavum yōcikkutēmaṉatu
             cirattaiyeṉumvāḷumutavip
   pantamaṟameyññāṉa tīramuntanteṉaip
             pātukāttaruḷceykuvāy
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (5)
   pūtalayamākiṉṟa māyaimutaleṉparcilar
             poṟipulaṉaṭaṅkumiṭamē
   poruḷeṉparcilarkaraṇa muṭiveṉparcilarkuṇm
             pōṉaviṭameṉparcilapēr
   nātavaṭiveṉparcilar vintumayameṉparcilar
             naṭṭanaṭuvēyirunta
   nāmeṉparcilaruruva māmeṉparcilarkaruti
             nāṭilaruḷeṉparcilapēr
   pētamaṟavuyirkeṭṭa nilaiyameṉṟiṭuvarcilar
             pēciṉaruveṉparcilapēr
   piṉṉumuṉṉuṅkeṭṭa cūṉiyamateṉparcilar
             piṟavumēmoḻivarivaiyāṟ
   pātaracamāymaṉatu cañcalappaṭumalāṟ
             paramacukaniṣṭaipeṟumō
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē (6)
   antakārattaiyō rakamākkimiṉpōleṉ
             ṉaṟivaiccurukkiṉavarā
   ravvaṟivutāṉumē paṟṟiṉatupaṟṟā
             yaḻuntavuntalaimītilē
   contamāyeḻutap paṭittārmeyññāṉa
             cukaniṣṭaicērāmalē
   cōṟṟutturuttiyaic catameṉavumuṇṭuṇṭu
             tūṅkavaittavarārkolō
   tantaitāymutalāṉa vakilaprapañcan
             taṉaittantateṉatācaiyō
   taṉṉaiyēnōvaṉō piṟaraiyēnōvaṉō
             taṟkālamatainōvaṉō
   pantamāṉa tutanta viṉaiyaiyēnōvaṉō
             paramārttamētumaṟiyēṉ
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (7)
   vārātelāmoḻiya varuvaṉavelāmeyta
             maṉatucākṣiyatākavē
   maruvanilaitantatum vētāntacittānta
             marapucamaracamākavē
   pūrāyamāyuṇara vūkamatutantātum
             poyyuṭalainilaiyaṉṟeṉap
   pōtaneṟitantatuñ cācuvatavāṉanta
             pōkamēvīṭeṉṉavē
   nīrāḷamāyuruka vuḷḷaṉputantatu
             niṉṉataruḷiṉṉumiṉṉu
   niṉṉaiyētuṇaiyeṉṟa veṉṉaiyēkākkavoru
             niṉaivucaṟṟuṇṭākilō
   pārātiyaṟiyātamōṉamēyiṭaiviṭāp
             paṟṟākaniṟkavaruḷvāy
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (8)
   āḻāḻikaraiyiṉṟi niṟkavilaiyōkoṭiya
             vālamamutākavilaiyō.
   vakkaṭaliṉmītuvaṭa vaṉaṉiṟkavillaiyō
             vantarattakilakōṭi
   tāḻāmaṉilainiṟka villaiyōmēruvun
             taṉuvākavaḷaiyavilaiyō
   captamēkaṅkaḷum vacrataraṉāṇaiyiṟ
             cañcarittiṭavillaiyō
   vāḻātuvāḻavē yirāmaṉaṭiyāṟcilaiyu
             maṭamaṅkaiyākavilaiyō
   maṇimantramātiyāl vēṇṭucittikaḷulaka
             mārkkattilvaikkavilaiyō
   pāḻāṉaveṉmaṉaṅ kuviyavorutantiram
             paṇṇuvatuṉakkarumaiyō
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē. (9)
   ācaikkoraḷavillai yakilamellāṅkaṭṭi
             yāḷiṉuṅkaṭaṉmītilē
   yāṇaicelavēniṉaiva raḷakēcaṉikarāka
             vampoṉmikavaittapēru
   nēcitturacavāta vittaikkalaintiṭuvar
             neṭunāḷiruntapēru
   nilaiyākvēyiṉuṅ kāyakaṟpantēṭi
             neñcupuṇṇāvarellām
   yōcikkumvēḷaiyiṟ pacitīravuṇpatu
             muṟaṅkuvatumākamuṭiyu
   muṉṉatēpōtunā ṉāṉeṉakkuḷaṟiyē
             yoṉṟaiviṭṭoṉṟupaṟṟip
   pācakkaṭaṟkuḷē vīḻāmaṉmaṉataṟṟa
             paricuttanilaiyaiyaruḷvāy
   pārkkumiṭameṅkumoru nīkkamaṟaniṟaikiṉṟa
             paripūraṇāṉantamē (10)
       ------------

                            3. poruḷvaṇakkam

       nittiyamāy nirmalamāy niṭkaḷamāy
                 nirāmayamāy niṟaivāynīṅkāc
       cuttamumāyt tūramumāyc camīpamumāyt
                 turiyaniṟai cuṭarāyellām
       vaittiruntatārakamā yāṉantamayamāki
                 maṉavākkeṭṭāc
       citturuvāyniṉṟavoṉṟaic cukārampap
                 peruveḷiyaic cintaiceyvām. (1)
       yātumaṉaniṉaiyumanta niṉaivukku
                 niṉaivāki yātiṉpālum
       pētamaṟaniṉṟuyiruk kuyirāki
                 yaṉparukkē pērāṉantak
       kōtilamutūṟṟarumpikkuṇaṅkuṟiyoṉṟaṟat
                 taṉṉaik koṭuttukkāṭṭun
       tītilparāparamāṉa cittāntap
                 pēroḷiyaic cintaiceyvām. (2)
       peruveḷiyāyaim pūtam piṟappiṭamāyp
                 pēcāta periyamōṉam
       varumiṭamāy maṉamātikkeṭṭāta
                 pēriṉpa mayamāyñāṉak
       kuruvaruḷāṟkāṭṭiṭavu maṉparaikkōttaṟa
                 viḻuṅkik koṇṭappālun
       terivaritāykkalantatentap poruḷantap
                 poruḷiṉaiyāñ cintaiceyvām. (3)
       ikaparamumuyirkkuyirai yāṉeṉataṟṟa
             varuṟavai yentanāḷuñ
       cukaparipūraṇamāṉa nirālampa
             kōcarattait turiyavāḻvai
       yakamakiḻavaruntēṉai mukkaṉiyaik
             kaṟkaṇṭaiyamirtaināṭi
       mokumokeṉaviruviḻinīr muttiṟaippakkara
             malarkaṇ mukiḻttuniṟpām. (4)
       cātikulampiṟappiṟappup panta
             muttiyaruvuruvattaṉmaināma
       mētumiṉṟiyepporuṭku mevviṭattum
             pirivaṟaniṉ ṟiyakkañceyyuñ
       cōtiyaimāttūveḷiyai maṉataviḻa
             niṟaivāṉa turiyavāḻvait
       tītilparamāmporuḷait tiruvaruḷē
             niṉaivākac cintaiceyvām. (5)
       intiracālaṅkaṉavu kāṉaliṉīreṉavulaka
             memakkuttōṉṟac
       cantatamuñ ciṟparattālaḻiyāta
             taṟparattaic cārntuvāḻka
       puntimakiḻuṟanāḷun taṭaiyaṟavāṉanta
             veḷḷam polikaveṉṟē
       vantaruḷuṅkurumauṉi malarttāḷaiyaṉu
             tiṉamum vaḻuttalceyvām. (6)
       poruḷākakkaṇṭaporu ḷevaikku
             mutaṟporuḷākip pōtamākit
       teruḷākikkarutumaṉpar miṭitīrap
             parukavanta ceḻuntēṉāki
       yaruḷāṉōrkkakampuṟameṉ
             ṟuṉṉāta pūraṇavāṉantamāki
       yiruṭīraviḷaṅkuporu ḷiyātantapporuḷiṉai
             yāmiṟaiñci niṟpām (7)
       arumaṟaiyiṉcirapporuḷāy viṇṇavar
             māmuṉivarcitta rātiyāṉōr
       terivariyapūraṇamāyk kāraṇaṅkaṟpaṉai
             kaṭanta celvamākik
       karutariyamalariṉmaṇa meḷḷileṇṇeyuṭaluyir
             pōṟ kalantennāḷun
       turiyanaṭuvūṭirunta periya
             poruḷiyātataṉait toḻutalceyvām (8)
       viṇṇātipūtamellān taṉṉakattilaṭakki
             veṟuveḷiyāyñāṉak
       kaṇṇārak kaṇṭavaṉpar kaṇṇūṭēyāṉantak
             kaṭalāyvēṟoṉ
       ṟeṇṇātapaṭikkiraṅkit tāṉākacceytaruḷu
             miṟaiyēyuṉṟaṉ
       ṟaṇṇāruñcāntavaru ṭaṉainiṉaintukaramalarka
             ṭalaimēṟkoḷvām. (9)
       viṇṇiṟaintaveḷiyāyeṉ maṉaveḷiyiṟ
             kalantaṟivām veḷīyiṉūṭun
       taṇṉiṟaintapēramutāyc catāṉantamāṉa
             peruntakaiyē niṉpā
       luṇṇiṟaintapēraṉpā luḷḷuruki
             moḻikuḷaṟiyuvakaiyākik
       kaṇṇiṟaintapuṉalukuppak karamukiḻppa
             niṉṉaruḷaik karuttilvaippām (10)
       (vēṟu)
       ātiyantaṅkāṭṭāta mutalāyemmai
             yaṭimaikkā vaḷarnteṭuttavaṉṉaipōla
       nītiperuṅkuruvākimaṉavākkeṭṭā
             niccayamāyc coccamatāy nimalamāki
       vātamiṭuñ camayaneṟikkariyatāki
             mauṉattōrpāl veḷiyām vayaṅkāniṉṟa
       cōtiyaiyeṉṉuyirttuṇaiyai nāṭikkaṇṇīr
             coriyaviru karaṅkuvittut toḻutalceyvām. (11)
       akaravuyireḻuttaṉaittumākivēṟā
             yamarntateṉa vakilāṇṭa maṉaittumākip
       pakarvaṉa vellāmākiyallavākip
             paramākiccollariya pāṉmaiyākit
       tukaḷuṟu caṅkaṟpa vikaṟpaṅkaḷellān
             tōyāta vaṟivākic cuttamāki
       nikaril pacupatiyāṉa poruḷaināṭi
             neṭṭuyirttup pēraṉpāṉiṉaital ceyvām. (12)

                          4. ciṉmāyaṉantakuru

       aṅkaikoṭumalrtūvi yaṅkamatupuḷakippa
                vaṉpiṉālurukiviḻinī
          rāṟākavārāta muttiyiṉatāvēca
                vācaikkaṭaṟkuṇmūḻkic
       caṅkaracuyampuvē campuvēyeṉavumoḻi
                taḻutaḻuttiṭavaṇṅkuñ
          caṉmārkkaneṟiyilāt tuṉmārkkaṉēṉaiyun
                taṇṇaruḷkoṭuttāḷvaiyō
       tuṅkamikupakkuvac caṉakaṉmutaṉmuṉivōrka
                ṭoḻutarukilvīṟṟiruppac
           collariyaneṟiyaiyoru collāluṇarttiyē
                corūpāṉupūtikāṭṭic
       ceṅkamalapīṭamēṟ kallālaṭikkuḷvaḷar
                cittāntamuttimutalē

                          5. mauṉakuruvaṇakkam.

       ācainikaḷattiṉai nirttūḷipaṭavutaṟi
                yāṅkāramuḷaiyaiyeṟṟi
          yattuvitamatamāki matamāṟumāṟāka
                vaṅkaiyiṉlilāḻiyākkip
       pācaviruṭaṉṉiḻa leṉaccuḷittārttumēṟ
                pārttupparantamaṉataip
          pārittakavaḷamāyp pūrikkavuṇṭumuka
                paṭāmaṉṉamāyainūṟit
       tēcupeṟanīvaitta ciṉmuttirāṅkucac
                ceṅkaikkuḷēyaṭaṅkic
          ciṉmayāṉantacuka veḷḷampaṭintuniṉ
                ṟiruvaruṭpūrttiyāṉa
       vācamuṟucaṟcāra mīteṉṉaiyoruñāṉa
                mattakajameṉavaḷrttāy
          mntrakuruvēyōka tantrakuruvēmūlaṉ
                marapilvarumeḷaṉakuruvē. (1)
       aintuvakaiyākiṉṟa pūtamutaṉātamu
                maṭaṅkaveḷiyākaveḷicey
          taṟiyāmaiyaṟivāti piṟivākavaṟivārka
                ḷaṟivākaniṉṟanilaiyiṟ
       cintaiyaṟanilleṉṟu cummāviruttimēṟ
                ciṉmayāṉantaveḷḷan
          tēkkittiḷaittunā ṉatuvāyirukkanī
                ceycitramikanaṉṟukā
       ṇentai vavāṟparama kuruvāḻkavāḻavaru
                ḷiyanantimarapuvāḻka
          veṉṟaṭiyarmaṉamakiḻa vētākamattuṇi
                piraṇṭillaiyoṉṟeṉṉavē
       vantakuruvēvīṟu civañāṉacittineṟi
                mauṉōpatēcakuruvē
          mantrakuruvēyōka tantrakuruvēmūlaṉ
                marapilvarumauṉakuruvē. (2)
       ātikkanalkiṉava rārintamāyaikke
                ṉaṟiviṉṟiyiṭamillaiyō
          vantarappuṣpamuṅ kāṉaliṉṉīrumō
                ravacarattupayōkamō
       pōtittanilaiyaiyu mayakkutēyapayanāṉ
                pukkavaruṭōṟṟiṭāmaṟ
          poyyāṉavulakattai meyyāniṟuttiyeṉ
                puntikkuḷintrajālañ
       cātikkutēyitaṉai vellavumupāyanī
                tantaruḷvateṉṟupukalvāy
          caṇmatastāpaṉamum vētāntacittānta
                camaracanirvākanilaiyu
       mātikkoṭaṇṭap parappelāmaṟiyavē
                vantaruḷuñāṉakuruvē
          mantarakuruvēyōka tantrakuruvēmūlaṉ
                 marapilvarumauṉakuruvē (3)
       miṉṉaṉaiyapoyyuṭalai nilaiyeṉṟumaiyilaku
                viḻikoṇṭumaiyalpūṭṭu
          miṉṉārkaḷiṉpamē meyyeṉṟumvaḷarmāṭa
                mēlvīṭucorkameṉṟum
       poṉṉaiyaḻiyātuvaḷai poruḷeṉṟupōṟṟiyip
                poyvēṣamikutikāṭṭip
          poṟaiyaṟivutuṟavīti lātinaṟkuṇamelām
                pōkkilēpōkaviṭṭut
       taṉṉikarilōpāti pāḻmpēypiṭittiṭat
                taraṇimicailōkāyataṉ
          camayanaṭaicārāmal vētāntacittānta
                camaracacivāṉupūti
       maṉṉavorucoṟkoṇ ṭeṉaittaṭut tāṇṭaṉpi
                ṉvāḻvittañāṉakuruvē
          mantrakuruvēyōka tantrakuruvēmūlaṉ
                marapilvarumeḷaṉakuruvē. (4)
       pōṉakamirukkiṉṟa cālaiyiṭaivēṇṭuva
                pucittaṟkirukkumatupōṟ
          puruṣarpeṟutarmāti vētamuṭaṉākamam
                pukalumatiṉālāmpayaṉ
       ñāṉaneṟimukyaneṟi kāṭciyaṉumāṉamuta
                ṉāṉāvitaṅkaṭērntu
          nāṉāṉeṉakkuḷaṟu paṭaipuṭaipeyarttiṭavu
                nāṉkucātaṉamumōrntiṭ
       ṭāṉaneṟiyāñcariyai yāticōpāṉumuṟ
                ṟaṇupakṣacampupakṣa
          māmirukaṟpamu māyāticēvaiyu
                maṟintiraṇṭoṉṟeṉṉumōr
       māṉatavikaṟpamaṟa veṉṟuniṟpatunamatu
                marapeṉṟaparamakuruvē
          mantrakuruvēyōka tantrakuruvēmūlaṉ
                 marapilvarumeḷaṉakuruvē. (5)
       kallātavaṟivumēṟ kēḷātakēḷviyuṅ
                karuṇaiciṟitētumillāk
          kāṭciyuṅkolaikaḷavu kaṭkāmamāṭciyāyk
                kātalittiṭuneñcamum
       pollātapoymmoḻiyu mallātunaṉmaikaḷ
                poruntukuṇamētumaṟiyēṉ
          puruṣarvaṭivāṉatē yallātukaṉavilum
                puruṣārttamētumillē
       ṉellāmaṟintanī yaṟiyātataṉṟeṉak
                kevvaṇṇamuyvaṇṇamō
          viruḷaiyiruḷeṉṟavark koḷitārakampeṟu
                meṉakkuniṉṉaruṭārakam
       vallāṉeṉumpeya ruṉakkuḷḷatēyinta
                vañcakaṉaiyāḷaniṉaiyāy
          mantrakuruvēyōka tntrakuruvēmūlaṉ
                 marapilvarumeḷaṉakuruvē. (6)
       kāṉakamilaṅkupuli pacuvoṭukulāvuniṉ
                kaṇkāṇamatayāṉainī
          kaikāṭṭavuṅkaiyā ṉekiṭikkeṉapperiya
                kaṭṭaimikavēntivarumē
       pōṉakamamaintateṉa vakkāmatēṉuniṉ
                poṉṉaṭiyiṉiṉṟucolumē
          puvirājarkavirājar tavarājaṉeṉṟuṉaip
                pōṟṟijayapōṟṟiyeṉpār

       vākaṉamumaṇṇakamum vantetirvaṇaṅkiṭumuṉ
                makimaiyatucollaveḷitō
          mantrakuruvēyōka tntrakuruvēmūlaṉ
                 marapilvarumeḷaṉakuruvē. (7)
       carukucalapakṣaṇika ḷorukōṭiyallāṟ
                cakōrapakṣikaḷpōlavē
          tavaḷanilavoḻukamirta tāraiyuṇṭaḻiyāta
                taṉmaiyaraṉantakōṭi
       yiruviṉaikaḷaṟṟiravu pakaleṉpataṟiyāta
                vēkāntamōṉañāṉa
          viṉpaniṣṭaiyarkōṭi maṇimantracittinilai
                yeytiṉarkaḷkōṭicūḻak
       kurumaṇiyiḻaittiṭṭa ciṅkātaṉattiṉmicai
                keāluvīṟṟirukkuniṉṉaik
          kumpiṭṭaṉantamuṟai teṇṭaniṭṭeṉmaṉak
                kuṟaiyelāntīrumvaṇṇa
       marumalareṭuttuṉiru tāḷaiyarccikkaveṉai
                vāveṉṟaḻaippatennāṇ
          mantrakuruvēyēāka tantrakuruvēmūlaṉ
                marapilvarumeḷaṉakuruvē. (8)
       āṅkāramāṉakula vēṭavempēypāḻtta
                vāṇavattiṉumvatulitukā
          ṇaṟiviṉaimayakkiṭu naṭuvaṟiyaveāṭṭāti
                yāteāṉṟuteāṭiṉumatuvāyt
       tāṅkātumeāḻipēcu marikarappiramāti
                tammeāṭucamāṉameṉṉun
          taṭaiyaṟṟatērilañ curuvāṇipēālavē
                taṉṉilacaiyātuniṟku
       mīṅkāreṉakkunika reṉṉapratāpit
                tirāvaṇākāramāki
          yitayaveḷiyeṅkaṇun taṉṉaracunāṭucey
                tirukkumitaṉeāṭenēramum
       vāṅkāvilāvaṭimai pēārāṭamuṭiyumēā
                meḷaṉēāpatēcakuruvē
          mantrakuruvēyēāka tantrakuruvēmūlaṉ
                 marapilvarumeḷaṉakuruvē. (9)
       paṟṟuvekuvitamāki yoṉṟaiviṭṭoṉṟaṉaip
                paṟṟiyuḻalkirumipōlap
          pāḻñcintaipeṟṟanāṉ veḷiyākaniṉṉaruḷ
                pakarntumaṟiyēṉṟuvitamō
       ciṟṟaṟivataṉṟiyu mevarēṉumorumoḻi
                tiṭukkeṉṟuraittapōtu
          cintaiceviyākavē paṟaiyaṟaiyavutaraven
                tīneñcamaḷavaḷāva
       vuṟṟuṇaravuṇarvaṟṟuṉ maṟṟaveṟiyiṉarpōla

          muṇarvateppaṭiyiṉpa tuṉpañcamāṉamā
                yuṟuvateppaṭiyāyiṉu
       maṟṟeṉakkaiyanī coṉṉavoruvārttaiyiṉai
                malaiyilakkeṉanampiṉēṉ
          mantrakuruvēyōka tantrakuruvēmūlaṉ
                 marapilvarumeḷaṉakuruvē. (10)

                        6. karuṇākarakkaṭavuḷ

       nirkkuṇanirāmaya nirañcaṉanirālampa
                nirviṣayakaivalyamā
          niṣkaḷavacaṅkacañ calarakitanirvacaṉa
                nirttōntanittamukta
       taṟparavisvātīta vyōmaparipūraṇa
                catāṉantañāṉapakava
          campucivacaṅkara carvēcaveṉṟunāṉ
                carvakālamuniṉaivaṉō
       vaṟputavakōcara nivarttipeṟumaṉparuk
                kāṉantapūrttiyāṉa
          vattuvitaniccaya corūpacākṣātkāra
                vaṉupūtiyaṉucūtamuṅ
       kaṟpaṉaiyaṟakkāṇa mukkaṇuṭaṉvaṭaniḻaṟ
                kaṇṇūṭiruntakuruvē
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                karuṇākarakkaṭavuḷē. (1)
       maṇṇātiyaintoṭu puṟattiluḷakaruviyum
                vākkāticurōtrātiyum
          vaḷarkiṉṟacaptāti maṉamātikalaiyāti
                maṉṉucuttātiyuṭaṉē
       toṇṇūṟṟoṭāṟumaṟ ṟuḷḷaṉavumauṉiyāyc
                coṉṉavorucoṟkoṇṭatē
          tūveḷiyatāyakaṇ ṭāṉantacukavāri
                tōṟṟumataiyeṉcolluvēṉ
       paṇṇārumicaiyiṉoṭu pāṭippaṭittaruṭ
                pāṉmaineṟiniṉṟutavaṟāp
          pakkuvavicēṣarāy nekkunekkurukip
                paṇinteḻuntirukaikūppik
       kaṇṇāṟukaraipuraḷa niṉṟavaṉparaiyelāṅ
                kaiviṭākkāṭciyuṟavē
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṇākarakkaṭavuḷē. (2)
       ellāmuṉaṭimaiyē yellāmuṉuṭaimaiyē
                yellāmuṉuṭaiyaceyalē
          yeṅkaṇumviyāpinī yeṉṟucolumiyalpeṉ
                ṟirukkātivētamellāñ
       collāṉmuḻakkiyatu mikkavupakāramāc
                colliṟantavarumviṇṭu
          coṉṉavaiyumivainalla kuruvāṉapērun
                tokuttaneṟitāṉumivaiyē
       yallāmalillaiyeṉa naṉṟāvaṟintē
                ṉaṟintapaṭiniṉṟucukanā
          ṉākātavaṇṇamē yivvaṇṇamāyiṉē
                ṉatuvuniṉataruḷeṉṉavē
       kallātvaṟiñaṉuk kuḷḷēyuṇarttiṉai
                katikkuvakaiyētupukalāy
          karutaṟiyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṇākarakkaṭavuḷē. (3)
       paṭṭappakaṟpoḻutai yiruḷeṉṟamaruḷartam
                pakṣamōveṉatupakṣama
          pārttaviṭameṅkaṇuṅ kōttanilaikulaiyātu
                paramaveḷiyākavorucoṟ
       ṟiṭṭamuṭaṉmeḷaṉiyā yaruḷceytirukkavuñ
                cērāmalārākanāṉ
          ciṟuvīṭukaṭṭiyati ṉaṭucōṟṟaiyuṇṭuṇṭu
                tēkkuciṟiyārkaḷpōla
       naṭṭaṇaiyatākkaṟṟa kalviyumvivēkamu
                naṉṉilayamākavuṉṉi
          nāṉeṉṟunīyeṉ ṟiraṇṭillaiyeṉṉavē
                naṭuvēmuḷaittamaṉataik
       kaṭṭavaṟiyāmalē vāṭiṉēṉeppōtu
                karuṇaikkurittāvaṉō
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṇākarakkaṭavuḷē. (4)
       meyviṭānāvuḷḷa meyyaruḷiruntunī
                meyyāṉameyyaiyellā
          meyyeṉavuṇarttiyatu meyyitaṟkaiyamilai
                meyyētumaṟiyāveṟum
       poyviṭāppoyyiṉē ṉuḷḷattiruntutāṉ
                poyyāṉapoyyaiyellām
          poyyeṉāvaṇṇamē pukalavaittāṉeṉiṟ
                puṉmaiyēṉeṉceykuveṉ
       maiviṭāteḻunīla kaṇṭakuruvēviṣṇu
                vaṭivāṉañāṉakuruvē
          malarmēvimaṟaiyōtu nāṉmukakkuruvē
                mataṅkaṭoṟuniṉṟakuruvē
       kaiviṭātēyeṉṟa vaṉparukkaṉpāyk
                karuttūṭuṇarttukuruvē
          karutaṟiyaciṟcapaiyi lāṉantanittamiṭu
                 karuṇākarakkaṭavuḷē. (5)
       paṇṇēṉuṉakkāṉa pūcaiyoruvaṭivilē
                pāvittiṟaiñcavāṅkē
          pārkkiṉṟamalarūṭu nīyēyiruttiyap
                paṉimalareṭukkamaṉamu
       naṇṇēṉalāmaliru kaitāṉkuvikkaveṉi
                ṉāṇumeṉṉuḷaniṟṟinī
          nāṉkumpiṭumpō taraikkumpiṭātalā
                ṉāṉpūcaiceyyaṉmuṟaiyō
       viṇṇēviṇātiyām pūtamēnātamē
                vētamēvētāntamē
          mētakkakēḷviyē kēḷviyāmpūmikkuḷ
                vittēyavittiṉmuḷaiyē
       kaṇṇēkaruttēye ṉeṇṇēyeḻuttē
                catikkāṉamōṉavaṭivē
          karutaṟiyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṇākarakkaṭavuḷē. (6)
       cantatamumvētamoḻi yātoṉṟupaṟṟiṉatu
                tāṉvantumuṟṟumeṉalāṟ
          cakamītiruntālu maraṇamuṇṭeṉpatu
                catāniṣṭarniṉaivatillai
       cintaiyaṟiyārkkītu pōtippatallavē
                ceppiṉumvekutarkkamān
          tivyakuṇamārkkaṇṭar cukarātimuṉivōrkaḷ
                cittāntanityaralavō
       vintrātitēvataikaḷ piramātikaṭavuḷa
                rirukkātivētamuṉiva
          reṇṇariyakaṇnātar navanātacittarka
                ḷiravimatiyātiyōrkaḷ
       kantaruvarkiṉṉararkaṇ maṟṟaiyarkaḷiyāvaruṅ
                kaikuvittiṭuteyvamē
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                karuṇākarakkaṭavuḷē (7)
       tuḷḷumaṟiyāmaṉatu palikoṭuttēṉkarma
                tuṣṭatēvataikaḷillai
          turiyaniṟaicāntatē vataiyāmuṉakkē
                toḻumpaṉaṉpapiṣēkanī
       ruḷḷuṟaiyileṉṉāvi naivēttiyamprāṇa
                ṉōṅkumatitūpatīpa
          morukālamaṉṟitu catākālapūcaiyā
                 voppuvittēṉkaruṇaikūr
       teḷḷimaṟaivaṭiyiṭṭa vamutappiḻampē
                teḷintatēṉēcīṉiyē
          tivyaracamiyāvun tiraṇṭoḻukupākē
                teviṭṭātavāṉantamē
       kaḷḷaṉaṟivūṭumē meḷḷameḷaveḷīyāyk
                kalakkavarunallavuṟavē
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                karuṇākarakkaṭavuḷē. (8)
       uṭalkuḻaiyaveṉpelā nekkurukaviḻinīrka
                ḷūṟṟeṉavetumpiyūṟṟa
          vūcikāntattiṉaik kaṇṭaṇukalpōlavē
                yōruṟavumuṉṉiyuṉṉip
       paṭapaṭeṉaneñcam pataittuṇṇaṭukkuṟap
                pāṭiyāṭikkutittup
          paṉimatimukattilē nilavaṉaiyapuṉṉakai
                parappiyārttārtteḻuntu
       maṭalaviḻumalaraṉaiya kaivirittukkūppi
                vāṉēyavāṉiviṉpa
          maḻaiyēmaḻaittārai veḷḷamēnīṭūḻi
                vāḻiyeṉavāḻttiyēttuṅ
       kaṭaṉmaṭaitiṟantaṉaiya vaṉparaṉpukkeḷiyai
                kaṉṉeñcaṉukkeḷiyaiyō
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṉākarakkaṭavuḷē. (9)
       iṅkaṟṟapaṭiyaṅku meṉavaṟiyunallaṟiña
                rekkālamummutavuvā
          riṉcoṟavaṟārpoymai yāmiḻukkuraiyā
                riraṅkuvārkolaikaḷipayilār
       caṅkaṟpacittarava ruḷḷakkaruttiluṟai
                cākṣinīyikaparattuñ
          cantāṉakaṟpakat tēvāyiruntē
                camastaviṉpamumutavuvāy
       ciṅkattaiyotteṉap pāyavaruviṉaiyiṉaic
                cētikkavarucimpuḷē
          cintākulattimira makalavarupāṉuvē
                tīṉaṉēṉkaraiyēṟavē
       kaṅkaṟṟapērācai veḷḷattiṉvaḷararuṭ
                kakaṉavaṭṭakkappalē
          karutariyaciṟcapaiyi lāṉantanirttamiṭu
                 karuṇākarakkaṭavuḷē. (10)

                              7. cittarkaṇam

       tikkoṭutikantamum maṉavēkameṉṉavē
             ceṉṟōṭiyāṭivaruvīr
       cempoṉmakamēruvoṭu kuṇamēruveṉṉavē
             tikaḻturuvaṉaḷavaḷāvi
       yukramikucakratara ṉeṉṉaniṟpīrkaiyi
             luḻuntamiḻumācamaṉamā
       vōrēḻukaṭalaiyum parukavallīrintra
             ṉulakumayirāvatamumē
       kaikkeḷiyapantā veṭuttuviḷaiyāṭuvīr
             kakaṉavaṭṭattaiyellāṅ
       kaṭukiṭaiyiruttiyē yaṣṭakulaveṟpaiyuṅ
             kāṭṭuvīrmēlumēlu
       mikkacittikaḷelām vallanīraṭimaimuṉ
             viḷaṅkavarucittiyilaiyō
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (!)
       pāṭṭaḷitutaintuvaḷar kaṟpakanaṉīḻalaip
             pāriṉiṭaivaravaḻaippīr
       patmaniticaṅkaniti yirupāricattilum
             paṇiceyuntoḻilāḷarpōṟ
       kēṭṭatukoṭuttuvara niṟkavaippīrpiccai
             kēṭṭuppiḻaippōraiyuṅ
       kirīṭapatiyākkuvīr kaṟpāntaveḷḷamoru
             kēṇiyiṭaikuṟukavaippī
       rōṭṭiṉaiyeṭuttā yiratteṭṭumāṟṟāka
             voḷiviṭumpoṉṉākkuvī
       rurakaṉumiḷaippāṟa yōkataṇṭattilē
             yulakucumaiyākavaruḷāṉ
       mīṭṭiṭavumvallanī reṉmaṉakkallaiyaṉaṉ
             meḻukākkivaippataritō
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (2)
       pāroṭunaṉṉīrāti yoṉṟoṭoṉṟākavē
             paṟṟilayamākupōtu
       paraveḷiyiṉmaruvuvīr kaṟpāntaveḷḷam
             parantiṭiṉataṟkumītē
       nīriluṟaivaṇṭāyt tuvaṇṭucivayōkanilai
             niṟpīrvikaṟpamāki
       neṭiyamukilēḻum parantuvaruṣikkilō
             nilavumatimaṇṭalamatē
       yūreṉaviḷaṅkuvīr piramātimuṭivilviṭai
             yūrtiyaruḷālulavuvī
       rulakaṅkaḷkīḻmēla tākapperuṅkāṟ
             ṟulāviṉaṟṟāraṇaiyiṉāṉ
       mēruveṉavacaiyāma ṉiṟkavallīrumatu
             mētakkacittiyeḷitō
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (3)
       eṇṇariyapiṟavitaṉiṉ māṉiṭappiṟavitā
             ṉiyātiṉummaritaritukā
       ṇippiṟavitappiṉā leppiṟavivāykkumō
             vētuvarumōvaṟikilēṉ
       kaṇṇakaṉilattunā ṉuḷḷapoḻutēyaruṭ
             kakaṉavaṭṭattiṉiṉṟu
       kālūṉṟiniṉṟupoḻi yāṉantamukiloṭu
             kalantumatiyavacamuṟavē
       paṇṇuvatunaṉmaiyin nilaipatiyumaṭṭumē
             patiyāyiruntatēkap
       pavarikulaiyāmalē kaurikuṇṭaliyāyi
             paṇṇavitaṉaruḷiṉālē
       viṇṇilavumatiyamuta moḻiyātupoḻiyavē
             vēṇṭuvēṉumataṭimaināṉ
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (4)
       poytikaḻumulakanaṭai yeṉcolkēṉeṉ col
             kēṉ poḻutupōkkēteṉṉilō
       poyyuṭaṉimittam pucippukkalaintiṭal
             pucittapiṉkaṇṇuṟaṅkal
       kaitavamalāmalitu ceytavamatallavē
             kaṇkeṭṭapērkkumveḷiyāyk
       kaṇṭatituviṇṭitaik kaṇṭittuniṟṟalek
             kālamōvataiyaṟikilēṉ
       maitikaḻumukiliṉaṅ kuṭainiḻaṟṟiṭavaṭṭa
             varaiyiṉoṭucempoṉmēru
       mālvaraiyiṉmutukūṭum yōkataṇṭakkōl
             varaintucayavirutukāṭṭi
       meytikaḻumaṣṭāṅka yōkapūmikkuḷvaḷar
             vēntarēkuṇacāntarē
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (5)
       keca turakamutalāṉa caturaṅkamaṉamāti
             kēḷvi yiṉicaintuniṟpak
       keṭikoṇṭatalamāṟu mummaṇṭalattiluṅ
             kiḷḷākkucellamikka
       tecavitamatāyniṉṟa nātaṅkaḷōliṭac
             ciṅkācaṉātiparkaḷāyt
       tikkuttikantamum pūraṇamatikkuṭai
             tikaḻntiṭavacantakāla
       micaiyamalarmītuṟai maṇampōlavāṉanta
             mitayamēṟkoḷḷumvaṇṇa
       meṉṟaikkumaḻiyāta civarācayōkamā
             yintrātitēvarkaḷelām
       vijayajayajayaveṉṉa vācicolavēkolu
             virukkunumperumaiyeḷitō
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (6)
       āṇilēpeṇṇilē yeṉpōlavorupētai
             yakilattiṉmicaiyuḷḷatō
       vāṭiyakaṟaṅkupō lōṭiyuḻalcintaiyai
             yaṭakkiyorukaṇamēṉumyāṉ
       kāṇilēṉtiruvaruḷai yallātumauṉiyāyk
             kaṇmūṭiyōṭumūccaik
       kaṭṭikkalāmatiyai muṭṭavēmūlaveṅ
             kaṉaliṉaiyeḻuppaniṉaivum
       pūṇilēṉiṟṟaināṭ kaṟṟatuṅkēṭṭatum
             pōkkilēpōkaviṭṭup
       poyyulakaṉāyiṉē ṉāyiṉuṅkaṭaiyāṉa
             puṉmaiyēṉiṉṉamiṉṉam
       vīṇilēyalaiyāmaṉ malaiyilakkākanīr
             veḷippaṭattōṟṟalvēṇṭum
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakccittarkaṇamē. (7)
       kaṉṉalamuteṉavumuk kaṉiyeṉavumvāyūṟu
             kaṇṭeṉavumaṭiyeṭuttuk
       kaṭavuḷarkaṭantatala vaḻutaḻutupēypōṟ
             karuttileḻukiṉṟatellā
       meṉṉataṟiyāmaiyaṟi veṉṉumirupakutiyā
             līṭṭutamiḻeṉṟamiḻiṉuk
       kiṉṉalpakarātulaka mārāmaimēliṭ
             ṭiruttalālittamiḻaiyē
       coṉṉavaṉiyāvaṉavaṉ mutticittikaḷelān
             tōyntaneṟiyēpaṭittīr
       collumeṉavavarnīṅkaḷ coṉṉavavaiyiṟciṟitu
             tōyntakuṇacāntaṉeṉavē
       miṉṉalpeṟavēcolla vaccolkēṭṭaṭimaimaṉa
             mvikacippatentanāḷō
       vētāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (8)
       poṟpiṉoṭukaikālil vaḷḷukirpaṭaittalāṟ
             pōntiṭaiyoṭukkamuṟalām
       polivāṉaveṇṇīṟu pūciyēyaruḷkoṇṭu
             pūrittaveṇṇīrmaiyā
       leṟpaṭaviḷṅkukaka ṉattilimaiyāviḻi
             yicaintumēṉōkkamuṟalā
       liravupakaliruḷāṉa kaṉatantipaṭanūṟi
       yitayaṅkaḷittiṭutalāṟ
             paṟpalavitaṅkoṇṭa pulikalaiyiṉuriyatu
       paṭaittuppratāpamuṟalāṟ
             paṉiveyilkaḷpukutāma ṉeṭiyavāṉṟoṭarneṭiya
       parumaravaṉaṅkaḷārum
             veṟpiṉiṭaiyuṟaitalāṟ ṟavarācaciṅkameṉa
       mikkōrumaippukaḻvarkāṇ
             vētāntacittānta camaracanaṉṉilaipeṟṟa
       vittakaccittarkaṇamē. (9)
       kallātapērkaḷē nallavarkaṇallavarkaḷ
             kaṟṟumaṟivilātaveṉ
       karmattaiyeṉcolkēṉ matiyaiyeṉcollukēṉ
             kaivalyañāṉanīti
       nallōruraikkilō karmamukkiyameṉṟu
             nāṭṭuvēṉkarmamoruva
       ṉāṭṭiṉālōpaḻaiya ñāṉamukkiyameṉṟu
             naviluvēṉvaṭamoḻiyilē
       vallāṉoruttaṉvara vuntrāviṭattilē
             vantatāvivakarippēṉ
       vallatamiḻaṟañarvari ṉaṅṅaṉēvaṭamoḻiyiṉ
             vacaṉaṅkaḷciṟitupukalvēṉ
       vellāmalevaraiyu maruṭṭiviṭavakaivanta
             vittaiyeṉmuttitarumō
       vetāntacittānta camaracanaṉṉilaipeṟṟa
             vittakaccittarkaṇamē. (10)

                          8. āṉantamāṉaparam.

       kollāmaiyettaṉai kuṇakkēṭṭainīkkumak
             kuṇamoṉṟumoṉṟilēṉpāṟ
       kōramettaṉaipakṣa pātamettaṉaivaṉ
             kuṇaṅkaḷettaṉai koṭiyapāḻṅ
       kallāmaiyettaṉai yakantaiyettaṉaimaṉak
             kaḷḷamettaṉaiyuḷḷacaṟ
       kāriyañcolliṭiṉu maṟiyāmaiyettaṉai
             katikkeṉṟamaittavaruḷiṟ
       cellāmaiyettaṉai virtākoṣṭiyeṉṉilō
             celvatettaṉaimuyaṟci
       cintaiyettaṉaicalaṉa mintrajālampōṉṟa
             tēkattilvāñcaimutalā
       yallāmaiyettaṉai yamaittaṉaiyuṉakkaṭimai
             yāṉēṉivaikkumāḷō
       vaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (1)
       teruḷākamaruḷāki yuḻalumaṉamāymaṉañ
             cērntuvaḷarcittākiyac
       cittelāñcūḻntaciva cittāyvicitramāyt
             tiramākināṉāvitap
       poruḷākiyapporuḷai yaṟipoṟiyumākiyaim
             pulaṉumāyaimpūtamāyp
       puṟamumāyakamumāyt tūrañcamīpamāyp
             pōkkoṭuvarattumāki
       yiruḷākiyoḷiyāki naṉmaitīmaiyumāki
             yiṉṟākināḷaiyāki
       yeṉṟumāyoṉṟumāyp palavumāyāvumā
             yivaiyallavāyaniṉṉai
       yaruḷākiniṉṟavarka ḷaṟivatallāloruva
             raṟivataṟkeḷitākumō
       yaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (2)
       māṟupaṭutarkkan toṭukkavaṟivārcāṇ
             vayiṟṟiṉporuṭṭatāka
       maṇṭalamumviṇṭalamu moṉṟākimaṉatuḻala
             mālākiniṟkavaṟivār
       vēṟupaṭuvēṭaṅkaḷ koḷḷavaṟivāroṉṟai
             meṇameṇeṉṟakamvēṟatām
       vittaiyaṟivāremaip pōlavēcantaipōṉ
             meynnūlvirikkavaṟivār
       cīṟupulipōṟcīṟi mūccaippiṭittuviḻi
             cekkaccivakkavaṟivār
       tirameṉṟutantam matattaiyētāmatac
             ceykaikoṭumuḷaṟavaṟivā
       rāṟucamayaṅkaṭoṟum vēṟuvēṟākiviḷai
             yāṭumuṉaiyāvaraṟivā
       raṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (3)
       kāyilaiyutirntakaṉi carukupuṉaṉmaṇṭiya
             kaṭumpacitaṉakkaṭaittuṅ
       kārvaraiyiṉmuḻaiyiṟ karuṅkalpōlacaiyātu
             kaṇmūṭineṭitiruntun
       tīyiṉiṭaivaikiyun tōyamatiṉmūḻkiyun
             tēkaṅkaḷeṉpelumpāyt
       teriyaniṉṟuñceṉṉi mayirkaḷkūṭākkuruvi
             teṟṟaveyilūṭiruntum
       vāyuvaiyaṭakkiyu maṉatiṉaiyaṭakkiyu
             mauṉattilēyiruntu
       matimaṇṭalattilē kaṉalcellavamutuṇṭu
             vaṉamūṭiruntumaṟiña
       rāyumaṟaimuṭivāṉa varuṇāṭiṉāraṭimai
             yakilattaināṭaṉmuṟaiyō
       vaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (4)
       cuttamumacuttamun tukkacukapētamun
             tontamuṭaṉirttontamum
       stūlamoṭucūkṣmamu mācaiyunirācaiyuñ
             collumorucolliṉmuṭivum
       pettamoṭumuttiyum pāvamoṭapāvamum
             pētamoṭapētanilaiyum
       perumaiyoṭuciṟumaiyu marumaiyuṭaṉeḷimaiyum
             peṇṇiṉuṭaṉāṇumaṟṟu
       nittamumaṉittamu mañcaṉanirañcaṉamu
             niṣkaḷamunikaḻcakaḷamu
       nītiyumaṉītiyu mātiyoṭaṉātiyu
             nirviṣayaviṣayavaṭivu
       mattaṉaiyunīyalate ḷattaṉaiyumillaiyeṉiṉ
             yāṅkaḷuṉaiyaṉṟiyuṇṭō
       vaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (5)
       kārārumāṇavak kāṭṭaikkaḷaintaṟak
             kaṇṭakaṅkārameṉṉuṅ
       kallaippiḷantuneñ cakamāṉapūmiveḷi
             kāṇattiruttimēṉmēṟ
       pārātiyaṟiyāta mōṉamāmvittaip
             patittaṉpunīrākavē
       pāycciyatupayirāku maṭṭumāmāyaivaṉ
             paṟavaiyaṇukātavaṇṇa
       nērākaniṉṟuviḷai pōkampucittuṇṭa
             niṉṉaṉparkūṭṭameyta
       niṉaiviṉpaṭikkunī muṉṉiṉṟukāppatē
             niṉṉaruṭpārameṉṟu
       mārāṟumaṟiyāta cūtāṉaveḷiyilveḷi
             yākiṉṟaturiyamayamē
       yaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (6)
       vāṉātipūtamā yakilāṇṭakōṭiyāy
             malaiyākivaḷaikaṭalumāy
       matiyākiyiraviyāy maṟṟuḷavelāmāki
             vāṉkaruṇaiveḷḷamāki
       nāṉākiniṉṟavaṉu nīyākiniṉṟiṭavu
             nāṉeṉpataṟṟiṭātē
       nāṉāṉeṉakkuḷaṟi nāṉāvikāriyāy
             nāṉaṟintaṟiyāmaiyāyp
       pōṉālatiṭṭavali vellaveḷitōpakaṟ
             poḻutupukumuṉkaṇmūṭip
       poyttuyilkoḷvāṉṟaṉai yeḻuppavacamōviṉip
             pōtippatentaneṟiyai
       yāṉālumeṉkoṭumai yaṉiyāyamaṉiyāya
             mārpāleṭuttumoḻivē
       ṉaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (7)
       poyyiṉēṉpulaiyiṉēṉ kolaiyiṉēṉiṉṉaruḷ
             pulappaṭavaṟintunilaiyāp
       puṉmaiyēṉkallāta taṉmaiyēṉaṉmaipōṟ
             poruḷalāpporuḷaināṭum
       veyyaṉēṉvekuḷiyēṉ veṟiyaṉēṉciṟiyaṉēṉ
             viṉaiyiṉēṉeṉṟeṉṉainī
       viṭṭuviṭaniṉaivaiyēṟ ṟaṭṭaḻivatallātu
             vēṟukatiyētupukalāy
       tuyyaṉēmeyyaṉē yuyiriṉukkuyirāṉa
             tuṇaivaṉēyiṇaiyoṉṟilāt
       turiyaṉēturiyamuṅ kāṇāvatītaṉē
             curutimuṭimītirunta
       vaiyaṉēyappaṉē yeṉumaṟiñaraṟivaiviṭ
             ṭakalātakaruṇaivaṭivē
       yaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (8)
       ettaṉaivitaṅkaṭāṉ kaṟkiṉuṅkēṭkiṉumeṉ
             ṉitayamumoṭuṅkavillai
       yāṉeṉumakantaitā ṉeḷḷaḷavumāṟavilai
             yātiṉummapimāṉameṉ
       cittamicaikuṭikoṇṭa tīkaiyōṭirakkameṉ
             ceṉmattunāṉaṟikilēṉ
       cīlamoṭutavavirata morukaṉavilāyiṉun
             taricaṉaṅkaṇṭumaṟiyēṉ
       poyttamoḻiyallāṉ maruntukkumeymmoḻi
             pukaṉṟiṭēṉpiṟarkēṭkavē
       pōtippatallātu cummāviruntaruḷ
             poruntiṭāppētaināṉē
       yattaṉaikuṇakkēṭar kaṇṭatākkēṭṭatā
             vavaṉimicaiyuṇṭōcolā
       yaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (9)
       ekkālamuntaṉak keṉṉavoruceyalilā
             vēḻainīyeṉṟiruntiṭ
       ṭeṉatāviyuṭalporuḷu meḷaṉiyāmvantukai
             yēṟṟunamateṉṟavaṉṟē
       poykkālatēcamum poypporuḷilvāñcaiyum
             poyyuṭalaimeyyeṉṉalum
       poyyuṟavupaṟṟalum poyyākunāṉeṉṉal
             poyyiṉumpoyyākaipāṉ
       maikkāliruṭṭaṉaiya viruḷillaiyiruviṉaikaḷ
             vantēṟavaḻiyumillai
       maṉamillaiyammaṉat tiṉamillaivēṟumoru
             varavillaipōkkumillai
       yakkālamikkāla meṉpatilaiyellā
             matītamayamāṉataṉṟō
       vaṇṭapakiraṇṭamu maṭaṅkavoruniṟaivāki
             yāṉantamāṉaparamē. (10)

                                9. cukavāri

       iṉṉamutukaṉipāku kaṟkaṇṭucīṉitē
             ṉeṉarucittiṭavaliyavan
       tiṉpaṅkoṭuttaniṉai yennēraniṉṉaṉpa
             riṭaiyaṟāturukināṭi
       yuṉṉiyakaruttaviḻa vuraikuḷaṟiyuṭaleṅku
             mōyntayarntavacamāki
       yuṇarvariyapēriṉpa vaṉupūtiyuṇarvilē
             yuṇarvārkaḷuḷḷapaṭikāṇ
       kaṉṉikaiyorutticiṟ ṟiṉpamvēmpeṉṉiṉuṅ
             kaikkoḷvaḷpakkuvattiṟ
       kaṇavaṉaruḷpeṟiṉmuṉē coṉṉavāṟeṉṉeṉak
             karutinakaiyāvaḷatupōṟ
       coṉṉapaṭikēṭkumip pētaikkuniṉkaruṇai
             tōṟṟiṟcukārampamāñ
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (1)
       aṉpiṉvaḻiyaṟiyāta veṉṉaittoṭarnteṉṉai
             yaṟiyātapakkuvattē
       yācaipperukkaip perukkikkoṭuttunā
             ṉaṟṟēṉalantēṉeṉa
       veṉpulamayaṅkavē pittēṟṟiviṭṭā
             yiraṅkiyoruvaḻiyāyiṉu
       miṉpaveḷamākavan tuḷḷaṅkaḷikkavē
             yeṉainīkalantatuṇṭō
       taṉparuvamalarukku maṇamuṇṭuvaṇṭuṇṭu
             taṇmukaitaṉakkumuṇṭō
       tamiyaṉēṟkivvaṇan tiruvuḷamiraṅkāta
             taṉmaiyāṟṟaṉiyiruntu
       tuṉpamuṟiṉeṅṅaṉē yaḻiyātaniṉṉaṉpar
             cukamvantuvāykkumuraiyāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (2)
       kallēṉumaiyavoru kālattilurukumeṉ
             kaṉṉeñcamurukavilaiyē
       karuṇaikkiṇaṅkāta vaṉmaiyaiyunāṉmukaṉ
             kaṟpikkavorukaṭavuḷō
       vallāṉvakuttatē vāykkāleṉumperu
             vaḻakkukkiḻukkumuṇṭō
       vāṉamāyniṉṟiṉpa maḻaiyāyiraṅkiyeṉai
             vāḻvippatuṉparaṅkāṇ
       pollātacēyeṉiṟ ṟāytaḷḷaṉītamō
             pukaliṭampiṟitumuṇṭō
       poyvārttaicollilō tiruvaruṭkayalumāyp
             puṉmaiyēṉāvaṉantō
       collāṉmuḻakkilō cukamillaimauṉiyāyc
             cummāvirukkavaruḷāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (3)
       eṉpelānekkuṭaiya rōmañcilirppavuṭa
             liḷakamaṉataḻaliṉmeḻukā
       yiṭaiyaṟāturukavaru maḻaipōliraṅkiyē
             yiruviḻikaṇīriṟaippa
       vaṉpiṉāṉmūrccitta vaṉparukkaṅṅaṉē
             yamirtacañcīvipōlvan
       tāṉantamaḻaipoḻivai yuḷḷaṉpilātaveṉai
             yārkkākavaṭimaikoṇṭāy
       puṉpulāṉmayirtō ṉarampeṉpumoyttiṭu
             pulaikkuṭililaruvaruppup
       poyyallavēyitaṉai meyyeṉṟunampiyeṉ
             punticelumōpāḻilē
       tuṉpamāyalaiyavō vulakanaṭaiyaiyavoru
             coṟpaṉattilumvēṇṭilēṉ
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (4)
       vennīrpoṟāteṉuṭal kāliṉmuṭṭaikkakavum
             veṭukkeṉṟacaitteṭuttāl
       viḻiyimaittaṅṅaṉē taṇṇaruḷaināṭuvēṉ
             vēṟoṉṟaiyoruvarcolli
       ṉannēramaiyōveṉ mukamvāṭiniṟpatuvu
             maiyaniṉṉaruḷaṟiyumē
       yāṉālumettap payantavaṉiyāṉeṉṉai
             yāṇṭanīkaiviṭātē
       yinnērameṉṟilai yuṭaṟcumaiyatākavu
             meṭuttāliṟakkaveṉṟē
       yeṅkeṅkumorutīrvai yāyamuṇṭāyiṉu
             miṟaiñcucukarātiyāṉa
       toṉṉīrmaiyāḷarkku māṉiṭamvakuttavaru
             ṭuṇaiyeṉṟunampukiṉṟēṉ
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (5)
       paṟṟuvaṉavaṟṟiṭu nirācaiyeṉṟorupūmi
             paṟṟippiṭikkumiyōkap
       pāṅkiṟpirāṇalaya meṉṉumorupūmiyivai
             paṟṟiṉmaṉamaṟumeṉṉavē
       kaṟṟaiyañcaṭaimauṉi tāṉēkaṉintakaṉi
             kaṉivikkavantakaṉipōṟ
       kaṇṭatinneṟiyeṉat tiruvuḷakkaṉiviṉoṭu
             kaṉivāytiṟantumoṉṟaip
       peṟṟavaṉumallēṉ peṟātavaṉumallēṉ
             perukkattavittuḷaṟiyē
       peṇṇīrmaiyeṉṉaviru kaṇṇīriṟaittunāṉ
             pēypōlirukkavulakañ
       cuṟṟinakaiceyyavē yulaiyaviṭṭāyeṉiṟ
             collaviṉivāyumuṇṭō
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (6)
       arumpoṉēmaṇiyē yeṉaṉpēyeṉaṉpāṉa
             vaṟivēyeṉaṟivilūṟu
       māṉantaveḷḷamē yeṉṟeṉṟupāṭiṉē
             ṉāṭiṉēṉāṭināṭi
       virumpiyēkūviṉē ṉulaṟiṉēṉalaṟiṉēṉ
             meycilirttirukaikūppi
       viṇmāriyeṉaveṉiru kaṇmāripeyyavē
             vēcaṟṟayarntēṉiyā
       ṉirumpunērneñcakak kaḷvaṉāṉālumuṉai
             yiṭaiviṭṭuniṉṟatuṇṭō
       veṉṟunīyaṉṟunā ṉuṉṉaṭimaiyallavō
             yātēṉumaṟiyāveṟun
       turumpaṉēṉeṉṉiṉuṅ kaiviṭutaṉītiyō
             toṇṭaroṭukūṭṭukaṇṭāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (7)
       pārātiyaṇṭaṅka ḷattaṉaiyumvaikkiṉṟa
             paraveḷiyiṉuṇmaikāṭṭip
       paṟṟumaṉaveḷikāṭṭi maṉaveḷiyiṉiṟṟōynta
             pāviyēṉparicukāṭṭit
       tārāḷamāyniṟka niccintaikāṭṭic
             catākālaniṣṭaiyeṉavē
       cakacanilaikāṭṭiṉai cukātītanilaiyan
             taṉaikkāṭṭanāḷcellumō
       kārāraveṇṇaru maṉantakōṭikaṇiṉṟu
             kālūṉṟimaḻaipoḻitalpōṟ
       kālvīcimiṉṉip paṭarntuparaveḷiyelāṅ
             kammiyāṉantaveḷḷañ
       cōrātupoḻiyavē karuṇaiyiṉmuḻaṅkiyē
             toṇṭaraikkūvumukilē
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (8)
       pētittacamayamō voṉṟucoṉapaṭiyoṉṟu
             pēcātutuṟavākiyē
       pēcātaperiyōrka ṇiruvikaṟpattiṉāṟ
             pēcārkaḷparamakuruvāyp
       pōtikkumukkaṇiṟai nērmaiyāykkaikkoṇṭu
             pōtippatāccaṟivilē
       pōkkuvaravaṟaviṉpa nīkkamaṟavacaṉamāp
             pōtippatevaraiyaṉē
       cātittacātaṉamu miyōkiyarkaṇamateṉṟu
             caṅkipparātalālē
       taṉṉilētāṉā yayarntuviṭuvōmeṉat
             taṉiyiruntiṭiṉaṅṅaṉē
       cōtikkamaṉamāyai taṉaiyēviṉālaṭimai
             cukamāvateppaṭicolāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (9)
       aṇṭamuṭitaṉṉilō pakiraṇṭamataṉilō
             valarimaṇṭalanaṭuvilō
       vaṉaṉaṭuvilōvamirta matinaṭuvilōvaṉpa
             rakamurukimalarkaṭūvit
       teṇṭamiṭavarumūrtti nilaiyilōtikkut
             tikantattilōveḷiyilō
       tikaḻvintunātanilai taṉṉilōvētānta
             cittāntanilaitaṉṉilō
       kaṇṭapalaporuḷilō kāṇātanilaiyeṉak
             kaṇṭacūṉiyamataṉilō
       kālamorumūṉṟilō piṟavinilaitaṉṉilō
             karuvikaraṇaṅkaḷōynta
       toṇṭarkaḷiṭattilō nīvīṟṟiruppatu
             toḻumpaṉēṟkuḷavupukalāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (10)
       entanāḷkaruṇaik kurittākunāḷeṉavu
             meṉṉitayameṉaivāṭṭutē
       yēteṉṟucolluvēṉ muṉṉoṭupiṉmalaivaṟavu
             miṟṟaivaraiyātu peṟṟēṉ
       pantamāṉatiliṭṭa meḻukākiyuḷḷam
             pataittuppataitturukavō
       paramacukamāvatu poṟuppariyatuyaramāyp
             palakālumūrccippatō
       cintaiyāṉatumaṟivai yeṉṉaṟivilaṟivāṉa
             teyvanīyaṉṟiyuḷatō
       tēkanilaiyallavē yuṭaikappalkappalāyt
             tiraiyāḻiyūṭucelumō
       contamāyāṇṭanī yaṟiyārkaḷpōlavē
             tuṉpattilāḻttaṉmuṟaiyō
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (11)
       ennāḷumuṭalilē yuyirāmuṉaippō
             lirukkavilaiyōmaṉateṉu
       miyāṉumeṉṉaṭpām pirāṇaṉumemaiccaṭama
             teṉṟuṉaiccitteṉṟumē
       yannāḷilevaṉō pirittāṉataikkēṭṭa
             vaṉṟumutaliṉṟuvaraiyu
       maniyāyamāyemai yaṭaikkikkuṟukkē
             yaṭarntaracupaṇṇiyeṅkaṇ
       muṉṉākanīyeṉṉa kōṭṭaikoṇṭāyeṉṟu
             mūṭamaṉamikavumēca
       mūṇṭeriyumaṉaliṭṭa meḻukāyuḷaṅkarukaṉ
             muṟaimaiyōpatiṉāyirañ
       coṉṉāluniṉṉaru ḷiraṅkavilaiyēyiṉic
             cukamvaruvateppaṭicolāy
       cuttanirkkuṇamāṉa parateyvamēparañ
             cōtiyēcukavāriyē. (12)
       --------

                      10. eṅkuniṟaikiṉṟaporuḷ

       avaṉaṉṟiyōraṇuvu macaiyāteṉumperiya
             vāpttarmoḻiyoṉṟukaṇṭā
       laṟivāvatētucila vaṟiyāmaiyētivai
             yaṟintārkaḷaṟiyārkaḷār
       mauṉamoṭiruntatā reṉpōluṭampelām
             vāyāyppitaṟṟumavarār
       maṉateṉavumorumāyai yeṅkēyiruntuvarum
             vaṉmaiyoṭirakkameṅkē
       puvaṉampaṭaippateṉ karttaviyamevviṭam
             pūtapētaṅkaḷeviṭam
       poymeyitamakitamēl varunaṉmaitīmaiyoṭu
             poṟaipoṟāmaiyumevviṭa
       mevarciṟiyarevarperiya revaruṟavarevarpakaiña
             riyātumuṉaiyaṉṟiyuṇṭō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (1)
       aṉṉēyaṉēyeṉuñ cilacamayaniṉṉaiyē
             yaiyāvaiyāveṉṉavē
       yalaṟiṭuñcilacamaya mallātupēypōla
             valaṟiyēyoṉṟumilavāyp
       piṉṉētumaṟiyāma loṉṟaiviṭṭoṉṟaip
             pitaṟṟiṭuñcilacamayamēṟ
       pēcariyavoḷiyeṉṟum veḷiyeṉṟunātāti
             piṟavumēnilayameṉṟun
       taṉṉērilātatō raṇuveṉṟumūvitat
             taṉmaiyāṅkālameṉṟuñ
       cāṟṟiṭuñcilacamaya mivaiyākivēṟatāyc
             catāñāṉavāṉantamā
       yeṉṉēyeṉēkaruṇai viḷaiyāṭṭiruntavā
             ṟemmaṉōrpukalaveḷitō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (2)
       vētamuṭaṉākama purāṇamitikācamutal
             vēṟumuḷakalaikaḷellā
       mikkākavattuvita tuvitamārkkattaiyē
             virivāveṭutturaikku
       mōtariyatuvitamē yattuvitañāṉattai
             yuṇṭupaṇuñāṉamāku
       mūkamaṉupavavacaṉa mūṉṟukkumovvumī
             tupayavātikaḷcammata
       mātaliṉeṉakkiṉic cariyaiyātikaḷpōtu
             miyātoṉṟupāvikkanā
       ṉatuvātalāluṉṉai nāṉeṉṟupāvikki
             ṉattuvitamārkkamuṟalā
       mētupāvittiṭiṉu matuvākivantaruḷce
             yentainīkuṟaiyumuṇṭō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (3)
       collāṉatiṟcaṟṟum vārātapiḷḷaiyait
             toṭṭilvaittāṭṭiyāṭṭit
       tuṭaiyiṉaikkiḷḷalpōṟ caṅkaṟpamoṉṟiṟ
             ṟoṭukkuntoṭuttaḻikkum
       pollātavātaṉai yeṉuñcaptapūmiyiṭai
             pōntutalaicuṟṟiyāṭum
       puruṣaṉilaṭaṅkāta pūvaipōṟṟāṉē
             puṟampōntucañcarikkuṅ
       kallōṭirumpukku mikavaṉmaikāṭṭiṭuṅ
             kāṇātukēṭṭavellāṅ
       kaṇṭatākkāṭṭiyē yaṇuvāccurikkiṭuṅ
             kapaṭanāṭakacālamō
       yellāmumvalatinta maṉamāyaiyēḻaiyā
             meṉṉālaṭakkavacamō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (4)
       kaṇṇāranīrmalki yuḷḷanekkurukāta
             kaḷḷaṉēṉāṉālumō
       kaikuvittāṭiyum pāṭiyumviṭāmalē
             kaṇpaṉittāraikāṭṭi
       yaṇṇāparañcōti yappāvuṉakkaṭimai
             yāṉeṉavumēleḻunta
       vaṉpākināṭaka naṭittatōkuṟaivillai
             yakilamuñciṟitaṟiyumēṟ
       ṟaṇṇāruniṉṉataru ḷaṟiyātatallavē
             caṟṟēṉumiṉitiraṅkic
       cācuvatamuttinilai yīteṉṟuṇarttiyē
             cakacanilaitantuvēṟoṉ
       ṟeṇṇāmaluḷḷapaṭi cukamāyirukkavē
             yēḻaiyēṟkaruḷceykaṇṭā
       yikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (5)
       kākamāṉatukōṭi kūṭukiṉṟālumoru
             kalliṉmuṉṉetirniṟkumō
       karmamāṉatukōṭi muṉṉēceytāluniṉ
             karuṇaipravākavaruḷait
       tākamāynāṭiṉarai vātikkavallatō
             tamiyaṉēṟkaruṭṭākamō
       caṟṟumilaiyeṉpatuvum veḷiyāccuviṉaiyelāñ
             caṅkētamāykkūṭiyē
       tēkamāṉataimikavum vāṭṭutētuṉpaṅkaḷ
             cērāmaliyōkamārkka
       cittiyōvaravillai cakacaniṣṭaikkumeṉ
             cintaikkumvekutūranā
       ṉēkamāyniṉṉō ṭirukkunāḷentanā
             ḷinnāḷiṉmuṟṟuṟātō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (6)
       orumaimaṉatākiyē yallalaṟaniṉṉaruḷi
             loruvaṉāṉvantirukki
       ṉulakampoṟātatō māyāvicitrameṉa
             vōyumōviṭamillaiyō
       varuḷuṭaiyaniṉṉaṉpar caṅkaiceytiṭuvarō
             valatukirttiyakarttarā
       yakilampaṭaittemmai yāḷkiṉṟapērcila
             raṭāteṉparōvakaṉṟa
       perumaipeṟupūraṇaṅ kuṟaiyumōpūtaṅkaḷ
             pēykkōlamāyvitaṇṭai
       pēcumōvalatutāṉ paripākakālam
             piṟakkavilaiyōtollaiyā
       mirumaiceṟicaṭaviṉai yetirttuvāypēcumō
             vētuḷavuciṟitupukalā
       yikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (7)
       nillātutēkameṉu niṉaivuṇṭutēkanilai
             niṉṟiṭavumauṉiyāki
       nērēyupāyamoṉ ṟaruḷiṉaiyaiyōvitaṉai
             niṉṟaṉuṭṭikkaveṉṟāṟ
       kallātamaṉamō voṭuṅkiyuparatipeṟak
             kāṇavilaiyākaiyālē
       kaiyēṟṟuṇumpucip povvātenāḷumuṉ
             kāṭciyiliruntukoṇṭu
       vallāḷarāyiyama niyamātimēṟkoṇṭa
             mātavarkkēvalceytu
       maṉatiṉpaṭikkelāñ cittipeṟalāñāṉam
             vāykkumorumaṉuveṉakkiṅ
       killāmaiyoṉṟiṉaiyu millāmaiyākavē
             yippōtiraṅkukaṇṭā
       yikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (8)
       maravuriyuṭuttumalai vaṉaneṟkoṟittumutir
             vaṉacarukuvāyilvantāl
       vaṉpacitavirttumaṉal veyilātimaḻaiyāl
             varuntiyummūlavaṉalaic
       ciramaḷaveḻuppiyu nīriṉiṭaimūḻkiyun
             tēkanamatallaveṉṟu
       ciṟcukavapēkṣaiyāy niṉṉaṉparyōkañ
             celuttiṉāriyāmpāviyēm
       viravumaṟucuvaiyiṉoṭu vēṇṭuvapucittaraiyil
             vēṇṭuvavelāmuṭutti
       mēṭaimāḷikaiyāti vīṭṭiṉiṭaivaikiyē
             vēṟoruvaruttamiṉṟi
       yiravupakalēḻaiyarkaḷ caiyōkamāyiṉō
             meppaṭipiḻaippaturaiyā
       yikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (9)
       muttaṉaiyamūralum pavaḷavāyiṉcolu
             mukattilakupacumañcaḷu
       mūrccikkavirakacaṉ ṉatamēṟṟavirukumpa
             mulaiyiṉmaṇimālaināla
       vaittemaimayakkiyiru kaṇvalaiyaivīciyē
             māyāvilācamōka
       vāritiyilāḻttiṭum pāḻāṉaciṟṟiṭai
             maṭantaiyarkaḷciṟṟiṉpamō
       puttamirtapōkam pucittuviḻiyimaiyāta
             poṉṉāṭṭumvantateṉṟāṟ
       pōrāṭṭamallavō pēriṉpamuttiyip
             pūmiyiliruntukāṇa
       vettaṉaivikātamvaru meṉṟucukarceṉṟaneṟi
             yivvulakamaṟiyātatō
       vikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (10)
       uṉṉilaiyumeṉṉilaiyu morunilaiyeṉakkiṭan
             tuḷaṟiṭumavattaiyāki
       yuṟavutāṉkāṭṭāta vāṇavamumoḷikaṇ
             ṭoḷikkiṉṟaviruḷeṉṉavē
       taṉṉilaimaikāṭṭā toruṅkaviruviṉaiyiṉāṟ
             ṟāvucukatukkavēlai
       taṭṭaḻiyamuṟṟumil lāmāyaiyataṉāṟ
             ṟaṭittakilapētamāṉa
       muṉṉilaiyoḻintiṭa vakaṇṭitākāramāy
             mūtaṟivumēlutippa
       muṉpiṉoṭukīḻmē ṉaṭuppakkameṉṉāmaṉ
             muṟṟumāṉantaniṟaivē
       yeṉṉilaimaiyāyniṟka viyalpukūraruḷvaṭiva
             mennāḷumvāḻivāḻi
       yikaparamiraṇṭiṉilu muyiriṉukkuyirāki
             yeṅkuniṟaikiṉṟaporuḷē. (10)
       --------

                          11. caccitāṉantacivam

       pāraratikakaṉap parappumuṇṭōveṉṟu
             paṭarveḷiyatākiyeḻunāp
       paritimatikāṇāc cuyañcōtiyāyaṇṭa
             pakiraṇṭavuyirevaikku
       nērākavaṟivā ykaṇṭamāyēkamāy
             nittamāynirttontamāy
       nirkkuṇavilācamāy vākkumaṉamaṇukāta
             nirmalāṉantamayamāyp
       pērātuniṟṟinī cummāviruntutāṉ
             pēriṉpameytiṭāmaṟ
       pēymaṉataiyaṇṭiyē tāyilāppiḷḷaipōṟ
             pittākavōmaṉataināṉ
       cārātapaṭiyaṟivi ṉiruvikaṟpāṅkamāñ
             cācuvataniṣṭaiyaruḷāy
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (1)
       kuṭakkoṭukuṇakkāti tikkiṉaiyuḻakkūṭu
             koḷḷalpōlaintupūtaṅ
       kūṭuñcuruṅkilaic cālēkamoṉpatu
             kulāvunaṭaimaṉaiyaināṟum
       vaṭakkayiṟuveṇṇaram pāveṉputacaiyiṉāṉ
             matavēḷviḻānaṭatta
       vaikkiṉṟakaittērai veṇṇīrcenīrkaṇir
             malanīrpuṇīriṟaikkum
       viṭakkutturuttiyaik karumaruntukkūṭṭai
             veṭṭaveṭṭattaḷirkkum
       vēṭkaimaramuṟukiṉṟa cuṭukāṭṭaimuṭivilē
             meypōliruntupoyyāñ
       caṭakkaiccaṭakkeṉac catameṉṟuciṉmayan
             tāṉākiniṟpateṉṟō
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (2)
       pākattiṉāṟkavitai pāṭippaṭikkavō
             pattineṟiyillaivēta
       pārāyaṇappaṉuvaṉ mūvarceypaṉuvalatu
             pakaravōvicaiyumillai
       yōkattilēciṟitu muyalaveṉṟāṟṟēka
             movvātivūṇveṟuttā
       luyirveṟuttiṭalokku mallātukiriyaika
             ḷupāyattiṉāṟceyyavō
       mōkattilēciṟitu moḻiyavilaimeyññāṉa
             mōṉattiṉiṟkaveṉṟāṉ
       muṟṟātuparipāka cattikaḷaṉēkaniṉ
             mūtaṟivilēyeḻunta
       tākattilēvāykku mamirtapravākamē
             taṉṉantaṉipperumaiyē
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (3)
       imaiyaḷavupōtaiyoru kaṟpakālampaṇṇu
             mivvulakamevvulakamō
       veṉṟeṇṇamvaruvikku mātarciṟṟiṉpamō
             veṉṉiṉmakamēruvākkic
       cumaiyeṭumiṉeṉṟutāṉ cummāṭumāyemaic
             cumaiyāḷumākkināḷun
       turpputtipaṇṇiyuḷa naṟputtiyāvaiyuñ
             cūṟaiyiṭṭintrajāla
       mamaiyavorukūttuñ camaintāṭumaṉamāyai
             yammammavellaleḷitō
       varuḷpeṟṟapērkkelā moḷipeṟṟuniṟkumī
             taruḷōvalātumaruḷō
       camaiyaneṟikāṇāta cākṣinīcūkṣmamāt
             tamiyaṉēṟkuḷavupukalāy
       carvaparipūraṇa vakaṇṭatattuvamāṉa       caccitāṉantacivamē. (4)
       iṉiyētemakkuṉaruḷ varumōveṉakkaruti
             yēṅkutēneñcamaiyō
       viṉṟaikkiruntārai nāḷaikkiruppāreṉ
             ṟeṇṇavōtiṭamillaiyē
       yaṉiyāyamāyinta vuṭalaināṉeṉṟuvaru
             mantakaṟkāḷākavō
       vāṭittirintunāṉ kaṟṟatuṅkēṭṭatu
             mavalamāyppōtaṉaṉṟō
       kaṉiyēṉumvaṟiyaceṅ kāyēṉumutircaruku
             kantamūlaṅkaḷēṉuṅ
       kaṉalvātaivanteyti ṉaḷḷippucittunāṉ
             kaṇmūṭimauṉiyākit
       taṉiyēyiruppataṟ keṇṇiṉēṉeṇṇamitu
             cāminīyaṟiyātatō
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (5)
       mattamatakarimukiṟ kulameṉṉaniṉṟilaku
             vāyiluṭaṉmatiyakaṭutōy
       māṭakūṭaccikara moyttacantirakānta
             maṇimēṭaiyuccimītu
       muttamiḻmuḻakkamuṭaṉ muttanakaiyārkaḷoṭu
             muttumuttāykkulāvi
       mōkattiruntumeṉ yōkattiṉilainiṉṟu
             mūccaippiṭittaṭaittuk
       kaittalanakappaṭai virittapuliciṅkamoṭu
             karaṭinuḻainūḻaikoṇṭa
       kāṉamalaiyucciyiṟ kukaiyūṭiruntumeṉ
             karatalāmalakameṉṉac
       cattamaṟamōṉanilai peṟṟavarkaḷuyvarkāṇ
             caṉakātituṇivitaṉṟō
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (6)
       kaittalamviḷaṅkumoru nelliyaṅkaṉiyeṉak
             kaṇṭavētākamattiṉ
       kāṭcipuruṣārttamatiṉ māṭcipeṟumuttiyatu
             karutiṉaṉumāṉamāti
       yuttipalavāniru vikaṟpamēlillaiyā
             loṉṟōṭiraṇṭeṉṉavō
       vuraiyumilainīyumilai nāṉumilaiyeṉpatu
             mupāyanīyuṇṭunāṉuñ
       cittamuḷaṉāṉillai yeṉumvacaṉanīyaṟivai
             teriyārkaṭeriyavacamō
       ceppukēvalanīti yoppuvamaiyallavē
             ciṉmuttirāṅkamarapiṟ
       cattamaṟaveṉaiyāṇṭa kurumauṉikaiyiṉāṟ
             ṟamiyaṉēṟkutavuporuḷē
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (7)
       kāyātamaramītu kallēṟucellumō
             kaṭavuṇīyāṅkaḷaṭiyēṅ
       karmapantattiṉāṟ caṉmapantampeṟak
             kaṟpittatuṉṉataruḷē
       vāyāravuṇṭapēr vāḻttuvatunontapēr
             vaivatuvumeṅkaḷulaka
       vāypāṭuniṟkaniṉ vaitikavoḻuṅkuniṉai
             vāḻttiṉāṟpeṟupēṟutā
       ṉōyātupeṟuvareṉa muṟaiyiṭṭatāṟpiṉṉa
             ruḷaṟuvatukarumamaṉṟā
       mupayaneṟiyīteṉṉi ṉucitaneṟiyentaneṟi
             yulakilēpiḻaipoṟukkun
       tāyāṉakaruṇaiyu muṉakkuṇṭeṉakkiṉic
             cañcalaṅkeṭavaruḷceyvāy
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (8)
       iṉṉampiṟappataṟ kiṭameṉṉiṉivvuṭala
             miṟavātiruppamūlat
       teḻumaṅkiyamirtoḻuku matimaṇṭalattiluṟa
             veṉṉammaikuṇṭaliṉipāṟ
       piṉṉampiṟakkātu cēyeṉavaḷarttiṭap
             pēyēṉainalkavēṇṭum
       piṟavātaneṟiyeṉak kuṇṭeṉṉīṉimmaiyē
             pēcukarppūratīpa
       miṉṉumpaṭikkakaṇ ṭākāravaṉṉaipāl
             viṉaiyēṉaiyoppuvittu
       vīṭṭuneṟikūṭṭiṭutaṉ mikavunaṉṟivaiyaṉṟi
             vivakāramuṇṭeṉṉilō
       taṉṉantaṉicciṟiya ṉāṟṟilēṉpōṟṟivaḷar
             caṉmārkkamuttimutalē
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (9)
       vētāvaiyivvaṇam vitittatēteṉṉiṉuṉ
             viṉaippakutiyeṉpaṉanta
       viṉaipēcavaṟiyātu niṟkavivaimaṉatāl
             viḷaintatāṉmaṉataināṭiṟ
       pōtamēniṟkumap pōtattaināṭilō
             pōtamuniṉālviḷakkam
       poyyaṉṟuteyvamaṟai yāvumēnīyeṉṟu
             pōkkuvaravaṟanikaḻttu
       mātāravātēya muḻutunīyātalā
             lakilamīteṉṉaiyāṭṭi
       yāṭalkaṇṭavaṉunī yāṭukiṉṟavaṉunī
             yaruḷunīmauṉañāṉa
       tātāvunīpeṟṟa tāytantaitāmunī
             tamarunīyāvunīkāṇ
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (10)
       kontaviḻmalarccōlai naṉṉīḻalvaikiṉuṅ
             kuḷirtīmpuṉaṟkaiyaḷḷik
       koḷḷukiṉumannī riṭaittiḷaittāṭiṉuṅ
             kuḷircantavāṭaimaṭavār
       vantulavukiṉṟateṉa muṉṟiliṭaiyulavavē
             vacatipeṟupōtumveḷḷai
       vaṭṭamatipaṭṭap pakaṟpōlanilavutara
             makiḻpōtumvēlaiyamutam
       vintaipeṟavaṟucuvaiyil vantateṉavamutuṇṇum
             vēḷaiyilumālaikantam
       veḷḷilaiyaṭaikkāy virumpivēṇṭiyavaṇṇam
             viḷaiyāṭiviḻituyiliṉuñ
       cantatamuniṉṉaruḷai maṟavāvarantantu
             tamiyēṉairakṣaipurivāy
       carvaparipūraṇa vakaṇṭatattuvamāṉa
             caccitāṉantacivamē. (11)
       ------------

                           12. tēcōmayāṉantam

       marumalarccōlaiceṟi naṉṉīḻaṉmalaiyāti
             maṉṉumuṉivarkkēvalāy
       mantramālikaicollu miyamaniyamātiyā
             mārkkattiṉiṉṟukoṇṭu
       karumaruvukāyattai nirmalamatākavē
             kamalācaṉāticērttuk
       kālaippiṭittaṉalai yammaikuṇṭaliyaṭik
             kalaimatiyiṉūṭutākki
       yurukivarumamirtattai yuṇṭuṇṭuṟaṅkāma
             luṇarvāṉaviḻiyaināṭi
       yoṉṟōṭiraṇṭeṉāc camaracacorūpacuka
             muṟṟiṭaveṉmaṉatiṉvaṇṇan
       tiruvaruṇmuṭikkavit tēkamoṭukāṇpaṉō
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (1)
       ippiṟaviyeṉṉumo riruṭkaṭaliṉmūḻkinā
             ṉeṉṉumorumakaravāypaṭ
       ṭiruviṉaiyeṉuntiraiyi ṉeṟṟuṇṭupuṟputa
             meṉakkoṅkaivaricaikāṭṭun
       tuppitaḻmaṭantaiyar mayaṟcaṇṭamārutac
             cuḻalvantuvantaṭippac
       cōrātavācaiyāṅ kāṉāṟuvāṉati
             curantateṉamēlumārppak
       kaipparicukārarpō laṟivāṉavaṅkamuṅ
             kaiviṭṭumatimayaṅkik
       kaḷḷavaṅkakkālar varuvareṉṟañciyē
             kaṇṇaruvikāṭṭumeḷiyēṉ
       ceppariyamuttiyāṅ karaicēravuṅkaruṇai
             ceyvaiyōcattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (2)
       tāytantaitamartāra makaveṉṉumivaiyelāñ
             cantaiyiṟkūṭṭamitilō
       cantēkamillaimaṇi māṭamāḷikaimēṭai
             caturaṅkacēṉaiyuṭaṉē
       vantatōrvāḻvumō rintracālakkōlam
             vañcaṉaipoṟāmailōpam
       vaittamaṉamāṅkirumi cērntamalapāṇṭamō
             vāñcaṉaiyilātakaṉavē
       yentanāḷuñcari yeṉattērntutērntumē
             yiravupakalillāviṭat
       tēkamāyniṉṟaniṉ ṉaruḷveḷḷamītilē
             yāṉeṉpataṟavumūḻkic
       cintaitāṉṟeḷiyātu cuḻalumvakaiyeṉkolō
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (3)
       āṭāmalōyntiṭṭa pamparampōlvicai
             yaṭaṅkimaṉamvīḻanērē
       yaṟiyāmaiyākiṉṟa viruḷakalaviruḷoḷiyu
             mallātiruntaveḷipōṟ
       kōṭāteṉaikkaṇ ṭeṉakkuṇiṟaicāntaveḷi
             kūṭiyiṉpātītamuṅ
       kūṭiṉēṉōcariyai kiriyaiyiṉmuyaṉṟuneṟi
             kūṭiṉēṉōvallaṉiyā
       ṉīṭākavēyāṟu vīṭṭiṉiṉirampiyē
             yilakivaḷarpirāṇaṉeṉṉu
       mirunitiyiṉaikkaṭṭi yōkaparaṉākāma
             lēḻaikkuṭumpaṉākit
       tēṭātaḻikkavoru mativantateṉkolō
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (4)
       pāṭātupāṭip paṭittaḷavilcamayamum
             pañcupaṭucollaṉivaṉaip
       pārmiṉōpārmiṉō veṉṟucapaikūṭavum
             paramārttamituveṉṉavē
       yāṭātumāṭineñ curukinekkāṭavē
             yamalamēyēkamēyem
       mātiyēcōtiyē yeṅkuniṟaikaṭavuḷē
             yaracēyeṉakkūvināṉ
       vāṭātuvāṭumeṉ mukavāṭṭamuṅkaṇṭu
             vāṭāveṉakkaruṇainī
       vaittiṭāvaṇṇamē caṅkētamāvinta
             vaṉmaiyaivaḷarppittatār
       tēṭātutēṭuvōr tēṭṭaṟṟatēṭṭamē
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (5)
       piṟiyātataṇṇaruṭ civañāṉiyāyvantu
             pēcariyavāciyālē
       pēriṉpavuṇmaiyai yaḷittaṉaiyeṉmaṉataṟap
             pērampalakkaṭavuḷā
       yaṟivāyiruntiṭu nātavolikāṭṭiyē
             yamirtapravākacitti
       yaruḷiṉaiyalātutiru vampalamumākiyeṉai
             yāṇṭaṉaipiṉeytineṟiyāyk
       kuṟitāṉaḷittaṉainaṉ maravurikoḷantaṇak
             kōlamāyacapānalaṅ
       kūṟiṉapiṉmauṉiyāyc cummāvirukkaneṟi
             kūṭṭiṉaiyelāmirukkac
       ciṟiyēṉmayaṅkimika vaṟiviṉmaiyāvaṉō
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (6)
       ārāreṉakkeṉṉa pōtittumeṉṉaveṉ
             ṉaṟiviṉaimayakkavacamō
       vaṇṭakōṭiyaiyelāṅ karuppavaṟaipōlavu
             maṭukkaṭukkāvamaittup
       pērāmaṉiṉṟapara veḷiyilēmaṉaveḷi
             piṟaṅkuvatalātoṉṟiṉum
       piṉṉamuṟamaruvātu naṉṉayattāliṉip
             pēriṉpamuttinilaiyun
       tārātutaḷḷavum pōkātuṉālatu
             taḷḷiṉumpōkēṉiyāṉ
       ṟaṭaiyētumillaiyāṇ ṭavaṉaṭimaiyeṉṉumiru
             taṉmaiyilumeṉvaḻakkut
       tīrātuviṭuvatilai naṭuvāṉakaṭavuḷē
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (7)
       kantukamatakkariyai vacamānaṭattalāṅ
             karaṭivempulivāyaiyuṅ
       kaṭṭalāmoruciṅka mutukiṉmēṟkoḷḷalāṅ
             kaṭceviyeṭuttāṭṭalām
       ventaḻaliṉiratamvait taintulōkattaiyum
             vētittuviṟṟuṇṇalām
       vēṟoruvarkāṇāma lulakattulāvalām
             viṇṇavaraiyēvalkoḷalāñ
       cantatamumiḷamaiyo ṭirukkalāmaṟṟoru
             carīrattiṉumpukutalāñ
       calamēṉaṭakkalāṅ kaṉaṉmēlirukkalān
             taṉṉikarilcittipeṟalāñ
       cintaiyaiyaṭakkiyē cummāvirukkiṉṟa
             tiṟamaritucattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (8)
       ellāmaṟintavaru mētumaṟiyātavaru
             millaiyeṉumivvulakamī
       tētumaṟiyātava ṉeṉappeyartarittumika
             vēḻaikkuḷēḻaiyākik
       kallātavaṟiviṟ kaṭaippaṭṭanāṉaṉṟu
             kaiyiṉāluṇmaiñāṉaṅ
       kaṟpittaniṉṉaruḷi ṉukkeṉṉakaimmāṟu
             kāṭṭuvēṉkuṟṟēvaṉā
       ṉallārntamēṉiyoṭu kuṇṭukaṭpiṟaiyeyiṟ
             ṟāpācavaṭivamāṉa
       vantakānīyoru pakaṭṭāṟpakaṭṭuva
             taṭātaṭākācunampāṟ
       cellātaṭāveṉṟu pēcavāyatutanta
             celvamēcattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (9)
       miṉpōlu miṭaiyoṭiyu moṭiyumeṉa moḻital
             pōṉ meṉcilampolikaḷārppa
       vīṅkippuṭaittuviḻu cumaiyaṉṉakoṅkaimaṭa
             miṉṉārkaḷpiṉṉēvalā
       leṉpōlalaintavarkaḷ kaṟṟārkaḷkallārka
             ḷiruvarkaḷiloruvaruṇṭō
       veṉceykēṉammamma veṉpāvameṉkoṭumai
             yēteṉṟeṭuttumoḻivē
       ṉaṉpālviyanturuki yaṭiyaṟṟamarameṉṉa
             vaṭiyilēvīḻntuvīḻntem
       maṭikaḷēyumataṭimai yāṅkaḷeṉunālvaruk
             kaṟamātiporuḷuraippat
       teṉpāliṉmukamāki vaṭavālirukkiṉṟa
             celvamēcattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (10)
       puttamirtapōkamuṅ kaṟpakanaṉīḻaliṟ
             polivuṟavirukkumiyalpum
       poṉṉulakilayirā vatattēṟuvaricaiyum
             pūmaṇṭalātikkamu
       mattaveṟiyiṉarvēṇṭu māleṉṟutaḷḷavume
             mālumorucuṭṭumaṟavē
       vaikkiṉṟavaippāḷaṉ mauṉatēcikaṉeṉṉa
             vantaniṉṉaruḷvāḻikāṇ
       cuttaparipūraṇa vakaṇṭamēyēkamē
             curutimuṭivāṉaporuḷē
       collariyavuyiriṉiṭai yaṅkaṅkuniṉṟaruḷ
             curantupoḻikaruṇaimukilē
       cittinilaimuttinilai viḷaikiṉṟapūmiyē
             tēṭariyacattākiyeṉ
       cittamicaikuṭikoṇṭa vaṟivāṉateyvamē
             tēcōmayāṉantamē. (11)

                         13. ciṟcukōtayavilācam

       kākamōṭukaḻu kalakaināynarikaḷ
             cuṟṟucōṟiṭuturuttiyaik
       kāliraṇṭunava vācalpeṟṟuvaḷar
             kāmavēṇaṭaṉacālaiyai
       mōkavācaimuṟi yiṭṭapeṭṭiyaimu
             malamikuntoḻukukēṇiyai
       moyttuveṅkirumi tattukumpiyai
             muṭaṅkalārkiṭaicarakkiṉai
       mākavintrataṉu miṉṉaiyottilaka
             vētamōtiyakulālaṉār
       vaṉaiyaveyyataṭi kāraṉāṉayamaṉ
             vantaṭikkumorumaṭkalat
       tēkamāṉapoyai meyyeṉakkaruti
             yaiyavaiyamicaivāṭavō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (1)
       kuṟikaḷōṭukuṇa mētumiṉṟiyaṉa
             loḻukaniṉṟiṭumirumpaṉaṟ
       kūṭaliṉṟiyatu vāyiruntapaṭi
             koṭiyavāṇavavaṟaikkuḷḷē
       yaṟivatētumaṟa vaṟivilāmaimaya
             māyirukkumeṉaiyaruḷiṉā
       laḷavilātataṉu karaṇamātiyai
             yaḷittapōtuṉaiyaṟintunāṉ
       piṟivilātavaṇa niṉṟiṭātapaṭi
             palaniṟaṅkavarumumalamāyp
       periyamāyaiyi laḻuntiniṉṉatu
             pracātanallaruṇmaṟantiṭuñ
       ciṟiyaṉēṉumuṉai vantaṇaintucuka
             māyiruppatiṉiyeṉṟukāṇ
       ṭerivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (2)
       aintupūtamoru kāṉaṉīreṉa
             vaṭaṅkavantaperuvāṉamē
       yātiyantanaṭu vētumiṉṟiyaru
             ḷāyniṟaintilakucōtiyē
       tontarūpamuṭa ṉarūpamātikuṟi
             kuṇamiṟantuvaḷarvastuvē
       turiyamēturiya vuyiriṉukkuṇarvu
             tōṉṟaniṉṟaruḷcupāvamē
       yentanāḷunaṭu vākiniṉṟoḷiru
             mātiyēkaruṇainītiyē
       yentaiyēyeṉa viṭaintiṭainturuku
             meḷiyaṉēṉkavalaitīravuñ
       cintaiyāṉatai yaṟintunīyuṉaruḷ
             ceyyanāṉumiṉiyuyvaṉō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (3)
       aivareṉṟapula vēṭarkoṭṭama
             taṭaṅkamarkkaṭavaṉmuṭṭiyā
       yaṭaviniṉṟumalai yarukiṉiṉṟucaru
             kātitiṉṟupaṉiveyiliṉāṉ
       meyvaruntutava millainaṟcariyai
             kiriyaiyōkameṉumūṉṟatāy
       mēvukiṉṟacavu pāṉanaṉṉeṟi
             virumpavillaiyulakattilē
       poymuṭaṅkutoḻil yātataṟkunala
             cāratittoḻiṉaṭattiṭum
       puttiyūkamaṟi vaṟṟamūkamivai
             poruḷeṉakkarutumaruḷaṉiyāṉ
       ṟeyvanallaruḷ paṭaittavaṉparoṭu
             cēravuṅkaruṇaikūrvaiyō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (4)
       ēkamāṉavuru vāṉanīyaruḷi
             ṉālaṉēkavuruvākiyē
       yentanāḷakila kōṭicirṣṭiceya
             vicaiyunāḷvaraiyanāṇmuta
       lākanāḷatu varaikkumuṉṉaṭimai
             kūṭavēcaṉaṉamāṉatō
       vaṉantamuṇṭunala caṉaṉamītitaṉu
             ḷaṟiyavēṇṭuvaṉavaṟiyalā
       mōkamātitaru pācamāṉatai
             yaṟintuviṭṭuṉaiyumeṉaiyumē
       muḻutuṇarntupara māṉaviṉpavau
             mūḻkavēṇṭumituviṉṟiyē
       tēkamēnaḻuvi nāṉumōnaḻuviṉ
             piṉṉaiyuyyumvakaiyuḷḷatō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (5)
       niyamalakṣaṇamu miyamalakṣaṇamu
             mācaṉātivitapētamu
       neṭituṇarntitaya patmapīṭamicai
             niṉṟilaṅkumacapāṉalat
       tiyalaṟintuvaḷar mūlakuṇṭaliyai
             yiṉitiṟaiñciyavaḷaruḷiṉā
       lellaiyaṟṟuvaḷar cēātimūlavaṉa
             leṅkaṇmōṉamaṉumuṟaiyilē
       vayamikuntuvaru mamirtamaṇṭala
             matikkuḷēmatiyaivaittunāṉ
       vāymaṭuttamirta vāriyaipparuki
             maṉṉumāramirtavaṭivamāyc
       ceyamikuntuvaru cittayōkanilai
             peṟṟuñāṉaneṟiyaṭaivaṉō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (6)
       eṟitiraikkaṭa ṉikarttacelvamika
             vallaleṉṟoruvarpiṉcelā
       tillaiyeṉṉumurai pēciṭātulaki
             levarumāmeṉamatikkavē
       neṟiyiṉvaikivaḷar celvamummutavi
             nōykaḷaṟṟacukavāḻkkaiyāy
       niyamamātinilai niṉṟuñāṉaneṟi
             niṣṭaikūṭavumennāḷumē
       yaṟiviṉiṉṟukuru vāyuṇarttiyatu
             maṉṟimōṉakuruvākiyē
       yakilamītuvara vantacīraruḷai
             yaiyavaiyaviṉiyeṉcolkēṉ
       ciṟiyaṉēḻainama taṭimaiyeṉṟuṉatu
             tiruvuḷattiṉiliruntatō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (7)
       evvuyirttiraḷu mulakileṉṉuyi
             reṉakkuḻainturukinaṉmaiyā
       yitamuraippaveṉa teṉṟayāvaiyu
             meṭutteṟintumatayāṉaipōṟ
       kavvaiyaṟṟanaṭai payilavaṉparaṭi
             kaṇṭatēvaruḷiṉvaṭivamāk
       kaṇṭayāvaiyu makaṇṭameṉṉaviru
             kaikuvittumalartūviyē
       pavvaveṇṭirai koṭuttataṇṭaraḷam
             viḻiyutirppamoḻikuḷaṟiyē
       pāṭiyāṭiyu ḷuṭaintuṭainteḻutu
             pāvaiyottacaitaliṉṟiyē
       tivyavaṉpuruva mākiyaṉparoṭu
             miṉpavīṭṭiṉiliruppaṉō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (8)
       mattarpēyaroṭu pālartaṉmaiyatu
             maruviyēturiyavaṭivamāy
       maṉṉutēcamoṭu kālamātiyai
             maṟantuniṉṉaṭiyaraṭiyilē
       pattiyāyneṭitu nampumeṉṉaiyoru
             maiyaṟantakilamāyaiyaip pārupāreṉa naṭattavantateṉa
             pāratattiṉumituḷḷatō
       cuttanittaviyal pākumōvuṉatu
             vicuvamāyainaṭuvākavē
       collavēṇumvakai nallakātikatai
             collumāyaiyiṉumillaiyeṉ
       cittamippaṭi mayaṅkumōvaruḷai
             nampiṉōrkaḷpeṟupēṟitō
       terivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (9)
       paṉmukaccamaya neṟipaṭaittavaru
             miyāṅkaḷēkaṭavuḷeṉṟiṭum
       pātakattavarum vātatarkkamiṭu
             paṭiṟaruntalaivaṇaṅkiṭat
       taṉmukattiluyir varavaḻaikkumema
             tarumaṉumpakaṭumēykkiyāyt
       taṉiyiruppavaṭa nīḻalūṭuvaḷar
             caṉakaṉātimuṉivōrkaṭañ
       coṉmayakkamatu tīravaṅkaikoṭu
             mōṉañāṉamatuṇarttiyē
       cuttanittavaru ḷiyalpatākavuḷa
             cōmacēkarakirpāṉuvāyt
       teṉmukattiṉmuka māyiruntakolu
             vemmukattiṉumvaṇaṅkuvēṉ
       ṟerivataṟkariya piramamēyamala
             ciṟcukōtayavilācamē. (10)
